    レッスン5: 非同期プログラミング（async/await）

    学習目標

    • 非同期プログラミングの概念と重要性を理解する
    • async/awaitキーワードの正しい使い方を習得する
    • Taskクラスを活用した並行処理を実装できるようになる
    • 非同期処理における例外処理とキャンセレーションを学ぶ

    1. 非同期プログラミングの基本概念

    1.1 同期処理 vs 非同期処理

    using System;
using System.Threading.Tasks;
using System.Threading;
using System.Net.Http;

public class AsyncBasics
{
    // 同期処理の例（ブロッキング）
    public static void SynchronousExample()
    {
        Console.WriteLine("同期処理開始");

        // 時間のかかる処理をシミュレート
        Thread.Sleep(3000); // 3秒間ブロック

        Console.WriteLine("同期処理完了");
        // この間、UIは固まったままになる
    }

    // 非同期処理の例（ノンブロッキング）
    public static async Task AsynchronousExample()
    {
        Console.WriteLine("非同期処理開始");

    {
        try
        {
            Console.WriteLine($"リクエスト開始: {url}");

            var response = await httpClient.GetAsync(url);
            response.EnsureSuccessStatusCode();

            var content = await response.Content.ReadAsStringAsync();

            Console.WriteLine($"レスポンス受信: {url} ({content.Length}文字)");
            return content;
        }
        catch (HttpRequestException ex)
        {
            Console.WriteLine($"HTTP要求エラー: {ex.Message}");

        catch (TaskCanceledException ex)
        {
            Console.WriteLine($"要求タイムアウト: {ex.Message}");
            return null;
        }
    }

    // 複数URLの並行取得
    public static async Task<Dictionary<string, string>> FetchMultipleUrlsAsync(string[] urls)
    {
        var results = new Dictionary<string, string>();

        // 各URLに対して非同期タスクを作成
        var fetchTasks = urls.Select(async url =>
        {
            var content = await FetchContentAsync(url);
            return new KeyValuePair<string, string>(url, content);
        });

        // すべてのタスクの完了を待機
        var completedTasks = await Task.WhenAll(fetchTasks);

        // 結果を辞書に格納
        foreach (var result in completedTasks)
        {
            results[result.Key] = result.Value;
        }

        return results;
    }

    // タイムアウト付きHTTPリクエスト
    public static async Task<string> FetchWithTimeoutAsync(string url, int timeoutSeconds)
    {
        using (var cts = new CancellationTokenSource(TimeSpan.FromSeconds(timeoutSeconds)))
        {
            try
            {
                var response = await httpClient.GetAsync(url, cts.Token);
                return await response.Content.ReadAsStringAsync();
            }
            catch (OperationCanceledException)
            {
                Console.WriteLine($"リクエストがタイムアウトしました: {url}");
                return null;
            }
        }
    }

    例題1: ファイルダウンロードマネージャー

    要求仕様： - 複数URLからファイルを並行ダウンロード - 進捗状況の表示 - キャンセル機能 - 失敗時のリトライ機能 - ダウンロード速度の制限

    解答例： 

    public class FileDownloadManager
{
    private readonly HttpClient httpClient;
    private readonly SemaphoreSlim semaphore;

    public FileDownloadManager(int maxConcurrentDownloads = 3)
    {
        httpClient = new HttpClient();
        semaphore = new SemaphoreSlim(maxConcurrentDownloads);
    }

    public class DownloadResult
    {
        public string Url { get; set; }
        public string LocalPath { get; set; }
        public bool IsSuccess { get; set; }
        public string ErrorMessage { get; set; }
        public long FileSize { get; set; }
        public TimeSpan Duration { get; set; }
    }

    public class DownloadProgress
    {
        public string Url { get; set; }
        public long BytesReceived { get; set; }
        public long TotalBytes { get; set; }
        public double ProgressPercentage => TotalBytes > 0 ? (BytesReceived * 100.0 / TotalBytes) : 0;
        public double SpeedBytesPerSecond { get; set; }
    }

    // 単一ファイルのダウンロード
    public async Task<DownloadResult> DownloadFileAsync(
        string url, 
        string localPath, 
        CancellationToken cancellationToken = default,
        IProgress<DownloadProgress> progress = null,
        int maxRetries = 3)
    {
        var startTime = DateTime.Now;
        var result = new DownloadResult
        {
            Url = url,
            LocalPath = localPath
        };

        for (int attempt = 1; attempt <= maxRetries; attempt++)
        {
            try
            {
                await semaphore.WaitAsync(cancellationToken);

                try
                {
                    await PerformDownloadAsync(url, localPath, cancellationToken, progress);

                    result.IsSuccess = true;
                    result.FileSize = new FileInfo(localPath).Length;
                    break;
                }
                finally
                {
                    semaphore.Release();
                }
            }
            catch (OperationCanceledException)
            {
                result.ErrorMessage = "ダウンロードがキャンセルされました";
                break;
            }
            catch (Exception ex)
            {
                result.ErrorMessage = ex.Message;

                if (attempt < maxRetries)
                {
                    Console.WriteLine($"ダウンロード失敗 (試行 {attempt}/{maxRetries}): {url}");
                    Console.WriteLine($"エラー: {ex.Message}");
                    Console.WriteLine($"3秒後にリトライします...");

                    await Task.Delay(3000, cancellationToken);
                }
                else
                {
                    Console.WriteLine($"ダウンロード最終失敗: {url}");
                }
            }
        }

        result.Duration = DateTime.Now - startTime;
        return result;
    }

    private async Task PerformDownloadAsync(
        string url, 
        string localPath, 
        CancellationToken cancellationToken,
        IProgress<DownloadProgress> progress)
    {
        using (var response = await httpClient.GetAsync(url, HttpCompletionOption.ResponseHeadersRead, cancellationToken))
        {
            response.EnsureSuccessStatusCode();

            var totalBytes = response.Content.Headers.ContentLength ?? 0;
            var progressInfo = new DownloadProgress
            {
                Url = url,
                TotalBytes = totalBytes
            };

            using (var contentStream = await response.Content.ReadAsStreamAsync())
            using (var fileStream = new FileStream(localPath, FileMode.Create, FileAccess.Write))
            {
                var buffer = new byte[8192];
                long totalBytesRead = 0;
                var stopwatch = Stopwatch.StartNew();

                while (true)
                {
                    var bytesRead = await contentStream.ReadAsync(buffer, 0, buffer.Length, cancellationToken);
                    if (bytesRead == 0) break;

                    await fileStream.WriteAsync(buffer, 0, bytesRead, cancellationToken);

                    totalBytesRead += bytesRead;
                    progressInfo.BytesReceived = totalBytesRead;

                    if (stopwatch.ElapsedMilliseconds > 0)
                    {
                        progressInfo.SpeedBytesPerSecond = totalBytesRead / (stopwatch.ElapsedMilliseconds / 1000.0);
                    }

                    progress?.Report(progressInfo);

                    // ダウンロード速度制限（オプション）
                    if (progressInfo.SpeedBytesPerSecond > 1024 * 1024) // 1MB/s制限
                    {
                        await Task.Delay(10, cancellationToken);
                    }
                }
            }
        }

        Console.WriteLine($"ダウンロード完了: {Path.GetFileName(localPath)}");
    }

    // 複数ファイルの並行ダウンロード
    public async Task<List<DownloadResult>> DownloadMultipleFilesAsync(
        Dictionary<string, string> urlToPathMap,
        CancellationToken cancellationToken = default,
        IProgress<string> overallProgress = null)
    {
        var downloadTasks = urlToPathMap.Select(async kvp =>
        {
            var progress = new Progress<DownloadProgress>(p =>
            {
                overallProgress?.Report($"{Path.GetFileName(kvp.Value)}: {p.ProgressPercentage:F1}% " +
                                      $"({FormatBytes(p.SpeedBytesPerSecond)}/s)");
            });

            return await DownloadFileAsync(kvp.Key, kvp.Value, cancellationToken, progress);
        });

        var results = await Task.WhenAll(downloadTasks);

        // 結果サマリーの表示
        var successCount = results.Count(r => r.IsSuccess);
        var totalSize = results.Where(r => r.IsSuccess).Sum(r => r.FileSize);
        var totalDuration = results.Max(r => r.Duration);

        Console.WriteLine($"\n=== ダウンロード完了サマリー ===");
        Console.WriteLine($"成功: {successCount}/{results.Length}");
        Console.WriteLine($"総サイズ: {FormatBytes(totalSize)}");
        Console.WriteLine($"総時間: {totalDuration:mm\\:ss}");

        if (totalDuration.TotalSeconds > 0)
        {
            var averageSpeed = totalSize / totalDuration.TotalSeconds;
            Console.WriteLine($"平均速度: {FormatBytes((long)averageSpeed)}/s");
        }

        return results.ToList();
    }

    // バッチダウンロードの管理
    public async Task<bool> BatchDownloadAsync(
        string[] urls, 
        string downloadDirectory,
        CancellationTokenSource cancellationTokenSource = null)
    {
        cancellationTokenSource ??= new CancellationTokenSource();

        if (!Directory.Exists(downloadDirectory))
            Directory.CreateDirectory(downloadDirectory);

        var urlToPathMap = urls.ToDictionary(
            url => url,
            url => Path.Combine(downloadDirectory, Path.GetFileName(new Uri(url).LocalPath))
        );

        var overallProgress = new Progress<string>(message =>
        {
            Console.WriteLine($"[{DateTime.Now:HH:mm:ss}] {message}");
        });

        try
        {
            var results = await DownloadMultipleFilesAsync(
                urlToPathMap, 
                cancellationTokenSource.Token, 
                overallProgress);

            var failedDownloads = results.Where(r => !r.IsSuccess).ToList();
            if (failedDownloads.Any())
            {
                Console.WriteLine("\n失敗したダウンロード:");
                foreach (var failed in failedDownloads)
                {
                    Console.WriteLine($"- {failed.Url}: {failed.ErrorMessage}");
                }
            }

            return results.All(r => r.IsSuccess);
        }
        catch (OperationCanceledException)
        {
            Console.WriteLine("バッチダウンロードがキャンセルされました");
            return false;
        }
    }

    private static string FormatBytes(long bytes)
    {
        string[] suffixes = { "B", "KB", "MB", "GB", "TB" };
        double size = bytes;
        int suffixIndex = 0;

        while (size >= 1024 && suffixIndex < suffixes.Length - 1)
        {
            size /= 1024;
            suffixIndex++;
        }

        return $"{size:F1} {suffixes[suffixIndex]}";
    }

    public void Dispose()
    {
        httpClient?.Dispose();
        semaphore?.Dispose();
    }
}
