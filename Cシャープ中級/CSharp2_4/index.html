    1. コレクションクラスの基本

    1.1 主要なコレクションクラス

    using System;
using System.Collections.Generic;
using System.Linq;

public class CollectionBasics
{
    public static void DemonstrateCollections()
    {
        // List<T> - 動的配列
        var numbers = new List<int> { 1, 2, 3, 4, 5 };
        numbers.Add(6);
        numbers.Remove(3);
        Console.WriteLine($"List要素数: {numbers.Count}");

        // Dictionary<TKey, TValue> - キー・値のペア
        var studentGrades = new Dictionary<string, int>
        {
            {"田中", 85},
            {"佐藤", 92},
            {"鈴木", 78}
        };
        studentGrades["田中"] = 88; // 値の更新
        studentGrades.Add("高橋", 95);

        // HashSet<T> - 重複を許さない集合
        var uniqueNumbers = new HashSet<int> { 1, 2, 3, 3, 4, 4, 5 };
        Console.WriteLine($"HashSet要素数: {uniqueNumbers.Count}"); // 5（重複除去）

        // Queue<T> - 先入先出（FIFO）
        var taskQueue = new Queue<string>();
        taskQueue.Enqueue("タスク1");

    public class Order

        public decimal Amount { get; set; }
    }

    public static void DemonstrateJoinsAndGrouping()
    {
        var customers = new List<Customer>
        {
            new Customer { Id = 1, Name = "田中太郎", City = "東京" },
            new Customer { Id = 2, Name = "佐藤花子", City = "大阪" },
            new Customer { Id = 3, Name = "鈴木次郎", City = "名古屋" },
            new Customer { Id = 4, Name = "高橋美咲", City = "東京" }
        };

        var orders = new List<Order>
        {
            new Order { Id = 1, CustomerId = 1, OrderDate = DateTime.Now.AddDays(-10), Amount = 15000 },
            new Order { Id = 2, CustomerId = 1, OrderDate = DateTime.Now.AddDays(-5), Amount = 8500 },
            new Order { Id = 3, CustomerId = 2, OrderDate = DateTime.Now.AddDays(-8), Amount = 22000 },
            new Order { Id = 4, CustomerId = 3, OrderDate = DateTime.Now.AddDays(-3), Amount = 12000 },
            new Order { Id = 5, CustomerId = 2, OrderDate = DateTime.Now.AddDays(-1), Amount = 5500 }
        };

        // Inner Join - 内部結合
        var customerOrders = from c in customers
                            join o in orders on c.Id equals o.CustomerId
                            select new
                            {
                                CustomerName = c.Name,
                                OrderId = o.Id,
                                OrderAmount = o.Amount,
                                OrderDate = o.OrderDate
                            };

        Console.WriteLine("顧客別注文一覧（内部結合）:");
        foreach (var co in customerOrders)
        {
            Console.WriteLine($"{co.CustomerName}: 注文#{co.OrderId} - ¥{co.OrderAmount:N0} ({co.OrderDate:yyyy/MM/dd})");
        }

        // Group Join - グループ結合
        var customersWithOrders = from c in customers
                                 join o in orders on c.Id equals o.CustomerId into customerOrders
                                 select new
                                 {
                                     Customer = c,
                                     Orders = customerOrders,
                                     TotalAmount = customerOrders.Sum(order => order.Amount),
                                     OrderCount = customerOrders.Count()
                                 };

        Console.WriteLine("\n顧客別注文サマリー:");
        foreach (var cwo in customersWithOrders)
        {
            Console.WriteLine($"{cwo.Customer.Name} ({cwo.Customer.City}):");
            Console.WriteLine($"  注文数: {cwo.OrderCount}, 総額: ¥{cwo.TotalAmount:N0}");
        }

        // 複雑なグルーピングと集約
        var citySummary = customers
            .Join(orders, c => c.Id, o => o.CustomerId, (c, o) => new { c.City, o.Amount })
            .GroupBy(x => x.City)
            .Select(g => new
            {
                City = g.Key,
                TotalSales = g.Sum(x => x.Amount),
                AverageSale = g.Average(x => x.Amount),
                OrderCount = g.Count()
            })
            .OrderByDescending(x => x.TotalSales);

        Console.WriteLine("\n都市別売上サマリー:");
        foreach (var summary in citySummary)
        {
            Console.WriteLine($"{summary.City}: 総売上¥{summary.TotalSales:N0}, " +
                            $"平均¥{summary.AverageSale:N0}, 注文数{summary.OrderCount}");
        }
    }
}

    3.2 条件処理とパフォーマンス最適化

    public class LinqOptimization
{
    public static void DemonstrateOptimization()
    {
        var numbers = Enumerable.Range(1, 1000000).ToList();

        // 遅延評価（Deferred Execution）
        var evenNumbers = numbers.Where(n => n % 2 == 0); // まだ実行されない
        var evenCount = evenNumbers.Count(); // ここで実行される

        // ToList()で即座に評価
        var evenNumbersList = numbers.Where(n => n % 2 == 0).ToList(); // 即座に実行

        // 効率的なクエリの書き方
        // ❌ 非効率：複数回の列挙
        var badExample = numbers.Where(n => n > 500000).Where(n => n % 2 == 0).Count();

        // ✅ 効率的：条件を結合
        var goodExample = numbers.Where(n => n > 500000 && n % 2 == 0).Count();

        // Any() と All() の活用
        var hasLargeNumbers = numbers.Any(n => n > 999000);
        var allPositive = numbers.All(n => n > 0);

        // First() vs FirstOrDefault()
        try
        {
            var firstEven = numbers.First(n => n % 2 == 0); // 見つからない場合は例外
            var firstOrDefault = numbers.FirstOrDefault(n => n > 2000000); // nullまたはdefault値
        }
        catch (InvalidOperationException)
        {
            Console.WriteLine("条件に一致する要素が見つかりませんでした");
        }

        // Skip() と Take() でページネーション
        int pageSize = 100;
        int pageNumber = 5;
        var pageData = numbers.Skip((pageNumber - 1) * pageSize).Take(pageSize);

        Console.WriteLine($"ページ{pageNumber}のデータ: {pageData.Count()}件");
    }
}

            .Where(n => n % 3 == 0 || n % 5 == 0)

                Count = acc.Count + 1,
                Max = Math.Max(acc.Max, n)
            });
