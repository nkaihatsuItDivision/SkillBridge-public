    1. 例外処理の基本概念

    1.1 例外とは

    例外（Exception）とは、プログラム実行中に発生する予期しない状況やエラーのことです。適切な例外処理により、プログラムの安定性と信頼性を向上させることができます。

    // 例外が発生する可能性があるコード例
public class BasicExceptionExample
{
    public static void DemonstrateExceptions()
    {
        // 1. ゼロ除算例外
        try
        {
            int result = 10 / 0; // DivideByZeroException
        }
        catch (DivideByZeroException ex)
        {
            Console.WriteLine($"ゼロ除算エラー: {ex.Message}");
        }

        // 2. null参照例外
        try
        {
            string text = null;
            int length = text.Length; // NullReferenceException
        }
        catch (NullReferenceException ex)
        {
            Console.WriteLine($"null参照エラー: {ex.Message}");
        }

        // 3. 配列の範囲外アクセス
        try
        {
            int[] numbers = { 1, 2, 3 };
            int value = numbers[5]; // IndexOutOfRangeException
        }
        catch (IndexOutOfRangeException ex)
        {
            Console.WriteLine($"配列範囲外エラー: {ex.Message}");
        }
    }
}

    1.2 try-catch-finally文の構造

    public class ExceptionHandlingStructure
{
    public static void HandleFileOperation(string fileName)
    {
        FileStream fileStream = null;

        try
        {
            // リスクのある処理
            fileStream = new FileStream(fileName, FileMode.Open);
            byte[] buffer = new byte[1024];
            int bytesRead = fileStream.Read(buffer, 0, buffer.Length);

            Console.WriteLine($"{bytesRead}バイト読み込みました");
        }
        catch (FileNotFoundException ex)
        {
            Console.WriteLine($"ファイルが見つかりません: {ex.FileName}");
        }
        catch (UnauthorizedAccessException ex)
        {
            Console.WriteLine($"ファイルへのアクセス権限がありません: {ex.Message}");
        }
        catch (IOException ex)
        {
            Console.WriteLine($"ファイル入出力エラー: {ex.Message}");
        }
        finally
        {
            // 必ず実行されるクリーンアップ処理
            fileStream?.Close();
            Console.WriteLine("ファイルストリームをクローズしました");
        }
    }
}

    2. カスタム例外クラス

    2.1 独自例外クラスの作成

    // カスタム例外クラス
public class InsufficientFundsException : Exception
{
    public decimal RequestedAmount { get; }
    public decimal AvailableBalance { get; }

    public InsufficientFundsException(decimal requestedAmount, decimal availableBalance)
        : base($"残高不足です。要求額: ¥{requestedAmount:N0}, 利用可能残高: ¥{availableBalance:N0}")
    {
        RequestedAmount = requestedAmount;
        AvailableBalance = availableBalance;
    }

    public InsufficientFundsException(decimal requestedAmount, decimal availableBalance, string message)
        : base(message)
    {
        RequestedAmount = requestedAmount;
        AvailableBalance = availableBalance;
    }

    public InsufficientFundsException(decimal requestedAmount, decimal availableBalance, string message, Exception innerException)
        : base(message, innerException)
    {
        RequestedAmount = requestedAmount;
        AvailableBalance = availableBalance;
    }
}

// 無効な口座番号例外
public class InvalidAccountNumberException : Exception

    public InvalidAccountNumberException(string accountNumber)

    2.2 カスタム例外を使用するクラス

    public class BankAccount
{
    private decimal balance;
    private string accountNumber;

    public string AccountNumber => accountNumber;
    public decimal Balance => balance;

    public BankAccount(string accountNumber, decimal initialBalance)
    {
        if (string.IsNullOrWhiteSpace(accountNumber) || accountNumber.Length != 10)
            throw new InvalidAccountNumberException(accountNumber);

        if (initialBalance < 0)
            throw new ArgumentException("初期残高は0以上である必要があります");

        this.accountNumber = accountNumber;
        this.balance = initialBalance;
    }

    public void Deposit(decimal amount)
    {
        if (amount <= 0)
            throw new ArgumentException("入金額は正の値である必要があります");

        try
        {
            checked // オーバーフロー検出を有効にする
            {
                balance += amount;
            }
        }
        catch (OverflowException)
        {
            throw new OverflowException("入金により残高が上限を超えます");
        }

        Console.WriteLine($"¥{amount:N0}を入金しました。現在の残高: ¥{balance:N0}");
    }

    public void Withdraw(decimal amount)
    {
        if (amount <= 0)
            throw new ArgumentException("出金額は正の値である必要があります");

        if (amount > balance)
            throw new InsufficientFundsException(amount, balance);

        balance -= amount;
        Console.WriteLine($"¥{amount:N0}を出金しました。現在の残高: ¥{balance:N0}");
    }

    public void Transfer(BankAccount destinationAccount, decimal amount)
    {
        if (destinationAccount == null)
            throw new ArgumentNullException(nameof(destinationAccount), "送金先口座が指定されていません");

        if (destinationAccount.AccountNumber == this.AccountNumber)
            throw new InvalidOperationException("同一口座への送金はできません");

        try
        {
            this.Withdraw(amount); // 出金（例外の可能性あり）
            destinationAccount.Deposit(amount); // 入金

            Console.WriteLine($"口座{destinationAccount.AccountNumber}へ¥{amount:N0}を送金しました");
        }
        catch (Exception)
        {
            // 出金に失敗した場合、何もしない（残高は変更されない）
            // 入金に失敗した場合、出金分を戻す必要があるが、簡略化のため省略
            throw; // 例外を再スロー
        }
    }
}

    3. 実践例題

    例題1: 学生成績管理システムの例外処理

    要求仕様： - 学生の成績を管理するクラス - 成績は0-100の範囲でなければならない - 学生IDは8桁の数字のみ有効 - 成績データの読み込み・保存時のエラーハンドリング - カスタム例外を適切に活用

    解答例： 

    // カスタム例外クラス群
public class InvalidScoreException : Exception
{
    public int AttemptedScore { get; }

    public InvalidScoreException(int score)
        : base($"無効な成績です: {score}（0-100の範囲で入力してください）")
    {
        AttemptedScore = score;
    }
}

public class InvalidStudentIdException : Exception
{
    public string AttemptedId { get; }

    public InvalidStudentIdException(string studentId)
        : base($"無効な学生IDです: {studentId}（8桁の数字で入力してください）")
    {
        AttemptedId = studentId;
    }
}

// 学生クラス
public class Student
{
    private Dictionary<string, int> scores;

    public string StudentId { get; private set; }
    public string Name { get; set; }

    public Student(string studentId, string name)
    {
        ValidateStudentId(studentId);

        StudentId = studentId;
        Name = name ?? throw new ArgumentNullException(nameof(name));
        scores = new Dictionary<string, int>();
    }

    private void ValidateStudentId(string studentId)
    {
        if (string.IsNullOrWhiteSpace(studentId) || 
            studentId.Length != 8 || 
            !studentId.All(char.IsDigit))
        {
            throw new InvalidStudentIdException(studentId);
        }
    }

    public void AddScore(string subject, int score)
    {
        if (string.IsNullOrWhiteSpace(subject))
            throw new ArgumentException("科目名が無効です");

        if (score < 0 || score > 100)
            throw new InvalidScoreException(score);

        scores[subject] = score;
        Console.WriteLine($"{Name}の{subject}の成績を{score}点で登録しました");
    }

    public int GetScore(string subject)
    {
        if (!scores.ContainsKey(subject))
            throw new KeyNotFoundException($"科目'{subject}'の成績が登録されていません");

        return scores[subject];
    }

    public double GetAverageScore()
    {
        if (scores.Count == 0)
            throw new InvalidOperationException("成績が登録されていないため平均を計算できません");

        return scores.Values.Average();
    }

    public void DisplayAllScores()
    {
        Console.WriteLine($"\n=== {Name}の成績一覧 ===");
        Console.WriteLine($"学生ID: {StudentId}");

        if (scores.Count == 0)
        {
            Console.WriteLine("登録された成績はありません");
            return;
        }

        foreach (var score in scores)
        {
            Console.WriteLine($"{score.Key}: {score.Value}点");
        }

        try
        {
            Console.WriteLine($"平均点: {GetAverageScore():F1}点");
        }
        catch (InvalidOperationException ex)
        {
            Console.WriteLine($"平均計算エラー: {ex.Message}");
        }
    }
}

// 成績管理システム
public class GradeManager
{
    private List<Student> students;
    private readonly string dataFilePath;

    public GradeManager(string dataFilePath = "grades.txt")
    {
        students = new List<Student>();
        this.dataFilePath = dataFilePath;
    }

    public void AddStudent(Student student)
    {
        if (student == null)
            throw new ArgumentNullException(nameof(student));

        if (students.Any(s => s.StudentId == student.StudentId))
            throw new InvalidOperationException($"学生ID {student.StudentId} は既に登録されています");

        students.Add(student);
        Console.WriteLine($"学生 {student.Name} を登録しました");
    }

    public Student FindStudent(string studentId)
    {
        var student = students.FirstOrDefault(s => s.StudentId == studentId);
        if (student == null)
            throw new KeyNotFoundException($"学生ID {studentId} が見つかりません");

        return student;
    }

    public void SaveToFile()
    {
        try
        {
            using (var writer = new StreamWriter(dataFilePath))
            {
                foreach (var student in students)
                {
                    writer.WriteLine($"{student.StudentId},{student.Name}");
                }
            }
            Console.WriteLine($"データを {dataFilePath} に保存しました");
        }
        catch (UnauthorizedAccessException ex)
        {
            throw new InvalidOperationException($"ファイルへの書き込み権限がありません: {dataFilePath}", ex);
        }
        catch (DirectoryNotFoundException ex)
        {
            throw new InvalidOperationException($"指定されたディレクトリが存在しません: {dataFilePath}", ex);
        }
        catch (IOException ex)
        {
            throw new InvalidOperationException($"ファイル保存中にエラーが発生しました: {ex.Message}", ex);
        }
    }

    public void LoadFromFile()
    {
        if (!File.Exists(dataFilePath))
        {
            Console.WriteLine($"データファイル {dataFilePath} が存在しません。新規作成します。");
            return;
        }

        try
        {
            using (var reader = new StreamReader(dataFilePath))
            {
                string line;
                int lineNumber = 0;

                while ((line = reader.ReadLine()) != null)
                {
                    lineNumber++;

                    try
                    {
                        var parts = line.Split(',');
                        if (parts.Length != 2)
                            throw new FormatException($"データ形式が正しくありません（行{lineNumber}）: {line}");

                        var student = new Student(parts[0].Trim(), parts[1].Trim());
                        students.Add(student);
                    }
                    catch (InvalidStudentIdException ex)
                    {
                        Console.WriteLine($"行{lineNumber}でエラー: {ex.Message}");
                    }
                    catch (FormatException ex)
                    {
                        Console.WriteLine($"行{lineNumber}でエラー: {ex.Message}");
                    }
                }
            }
            Console.WriteLine($"データを {dataFilePath} から読み込みました");
        }
        catch (IOException ex)
        {
            throw new InvalidOperationException($"ファイル読み込み中にエラーが発生しました: {ex.Message}", ex);
        }
    }

    public void DisplayAllStudents()
    {
        Console.WriteLine("\n=== 全学生一覧 ===");
        if (students.Count == 0)
        {
            Console.WriteLine("登録された学生はいません");
            return;
        }

        foreach (var student in students.OrderBy(s => s.StudentId))
        {
            student.DisplayAllScores();
        }
    }
}

    4.1 適切な例外処理のガイドライン

    public class BestPracticesExample
{
    // ✅ 良い例：具体的な例外をキャッチ
    public static string ReadConfigValue(string key)
    {
        try
        {
            var config = ConfigurationManager.AppSettings[key];
            if (string.IsNullOrEmpty(config))
                throw new ConfigurationErrorsException($"設定値 '{key}' が見つかりません");

            return config;
        }
        catch (ConfigurationErrorsException)
        {
            // 設定エラーは再スロー
            throw;
        }
        catch (Exception ex)
        {
            // 予期しないエラーをラップして詳細情報を追加
            throw new ApplicationException($"設定値の読み込み中にエラーが発生しました: {key}", ex);
        }
    }

    // ✅ 良い例：using文によるリソース管理
    public static void ProcessFile(string fileName)
    {
        try
        {
            using (var fileStream = new FileStream(fileName, FileMode.Open))
            using (var reader = new StreamReader(fileStream))
            {
                string content = reader.ReadToEnd();
                ProcessContent(content);
            }
            // using文により自動的にリソースが解放される
        }
        catch (FileNotFoundException)
        {
            Console.WriteLine($"ファイル '{fileName}' が見つかりません");
            throw;
        }
        catch (UnauthorizedAccessException)
        {
            Console.WriteLine($"ファイル '{fileName}' へのアクセス権限がありません");
            throw;
        }
    }

    private static void ProcessContent(string content)
    {
        // ファイル内容の処理
    }

    // ❌ 悪い例：Exception をキャッチして隠蔽
    public static void BadExample()
    {
        try
        {
            // 何らかの処理
            int result = 10 / 0;
        }
        catch (Exception)
        {
            // 例外を隠蔽（デバッグが困難になる）
            Console.WriteLine("エラーが発生しました");
        }
    }

    // ✅ 良い例：例外情報を適切にログ出力
    public static void GoodExample()
    {
        try
        {
            // 何らかの処理
            PerformOperation();
        }
        catch (ArgumentException ex)
        {
            Console.WriteLine($"引数エラー: {ex.Message}");
            // 必要に応じて再スロー
            throw;
        }
        catch (InvalidOperationException ex)
        {
            Console.WriteLine($"操作エラー: {ex.Message}");
            // ログに記録
            LogError(ex);
            throw;
        }
    }

    private static void PerformOperation()
    {
        // 処理の実装
    }

    private static void LogError(Exception ex)
    {
        // エラーログの実装
        Console.WriteLine($"[{DateTime.Now}] ERROR: {ex}");
    }
}

    例外処理は、堅牢なアプリケーションを作成するために不可欠な技術です。適切な例外処理により、予期しないエラーからアプリケーションを保護し、ユーザーに分かりやすいエラーメッセージを提供できます。カスタム例外を効果的に活用し、例外処理のベストプラクティスに従うことで、保守性の高いコードを書けるようになります。
