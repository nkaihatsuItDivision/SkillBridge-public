    Dictionaryオブジェクト

    🎯 このレッスンの目標

    • Dictionaryオブジェクトの概念を理解する
    • キーと値のペアでデータを管理できるようになる
    • 重複チェックができるようになる
    • Dictionaryを使った集計処理ができるようになる
    • 実務で使える活用パターンを習得する

    学習時間: 約70分

    📚 前回の復習

    前回のレッスンで学んだこと：

    • 静的配列・動的配列
    • ReDim / ReDim Preserve
    • セル範囲と配列の変換
    • 配列による高速処理

    今回は、配列よりも柔軟な「Dictionary」を学びます！

    1. Dictionaryとは？

    配列との違い

    配列：インデックス（番号）でデータにアクセス 

    data(0) = "りんご"
data(1) = "みかん"

    Dictionary：キー（名前）でデータにアクセス 

    dict("fruit1") = "りんご"
dict("fruit2") = "みかん"

    📘 Dictionaryのイメージ

    Dictionaryの特徴

    1. キーで直接アクセスできる
    2. キーの重複は許可されない
    3. 存在確認が高速
    4. 動的にサイズが変わる

    2. Dictionaryの作成

    CreateObjectを使う方法（推奨）

    Dim dict As Object
Set dict = CreateObject("Scripting.Dictionary")

    参照設定を使う方法

    1. VBE → ツール → 参照設定
    2. 「Microsoft Scripting Runtime」にチェック
    Dim dict As Scripting.Dictionary
Set dict = New Scripting.Dictionary

    推奨：CreateObjectを使う（参照設定不要で他のPCでも動作）

    3. 基本操作

    値の追加

    Option Explicit

Sub DictionaryBasic()
    Dim dict As Object
    Set dict = CreateObject("Scripting.Dictionary")

    ' 値を追加（Add メソッド）
    dict.Add "A001", "田中太郎"
    dict.Add "A002", "山田花子"
    dict.Add "A003", "鈴木一郎"

    ' 値を追加（直接代入）
    dict("A004") = "佐藤次郎"
End Sub

    AddとItem直接代入の違い

    方法 キーが既存の場合
    dict.Add key, value エラー発生
    dict(key) = value 値を上書き

    値の取得

    Dim name As String
name = dict("A001")  ' "田中太郎"
name = dict.Item("A001")  ' 同じ意味

    値の変更

    dict("A001") = "田中三郎"  ' 値を上書き

    値の削除

    dict.Remove "A001"  ' キー"A001"を削除
dict.RemoveAll       ' すべて削除

    4. 存在確認（Exists）

    6. 重複チェック

    実践例：重複データの検出

    Option Explicit

Sub FindDuplicates()
    Dim ws As Worksheet
    Dim dict As Object
    Dim lastRow As Long
    Dim i As Long
    Dim value As String
    Dim duplicates As String

    Set ws = ThisWorkbook.Worksheets("Sheet1")
    Set dict = CreateObject("Scripting.Dictionary")

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    duplicates = ""

    For i = 2 To lastRow
        value = ws.Cells(i, 1).Value

        If dict.Exists(value) Then
            ' 重複発見
            duplicates = duplicates & value & vbCrLf
        Else
            ' 新規登録
            dict.Add value, i  ' 値をキー、行番号を値として登録
        End If
    Next i

    If duplicates = "" Then
        MsgBox "重複はありません"
    Else
        MsgBox "重複データ:" & vbCrLf & duplicates
    End If
End Sub

Sub SumByCategory()

    9. 実践：クロス集計

    Option Explicit

Sub CrossTabulation()
    Dim ws As Worksheet
    Dim wsResult As Worksheet
    Dim dict As Object
    Dim data As Variant
    Dim lastRow As Long
    Dim i As Long
    Dim rowKey As String, colKey As String
    Dim key As Variant
    Dim rowKeys As Object, colKeys As Object

    Set ws = ThisWorkbook.Worksheets("データ")
    Set dict = CreateObject("Scripting.Dictionary")
    Set rowKeys = CreateObject("Scripting.Dictionary")
    Set colKeys = CreateObject("Scripting.Dictionary")

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    data = ws.Range("A2:C" & lastRow).Value

    ' 集計
    For i = 1 To UBound(data, 1)
        rowKey = data(i, 1)  ' 行キー（例：部署）
        colKey = data(i, 2)  ' 列キー（例：月）

        ' キーの組み合わせで集計
        key = rowKey & "|" & colKey

        If dict.Exists(key) Then
            dict(key) = dict(key) + data(i, 3)
        Else
            dict.Add key, data(i, 3)
        End If

        ' 行・列のキーを収集
        If Not rowKeys.Exists(rowKey) Then rowKeys.Add rowKey, ""
        If Not colKeys.Exists(colKey) Then colKeys.Add colKey, ""
    Next i

    MsgBox "集計完了！" & vbCrLf & _
           "行カテゴリ: " & rowKeys.Count & "種類" & vbCrLf & _
           "列カテゴリ: " & colKeys.Count & "種類"
End Sub

    課題1：出現回数のカウント

    A列のデータについて、各値が何回出現するかをカウントするマクロを作成してください。

    解答例 
    Sub CountOccurrences()
    Dim ws As Worksheet
    Dim dict As Object
    Dim lastRow As Long
    Dim i As Long
    Dim value As String
    Dim key As Variant

    Set ws = ActiveSheet
    Set dict = CreateObject("Scripting.Dictionary")

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    For i = 2 To lastRow
        value = ws.Cells(i, 1).Value
        If dict.Exists(value) Then
            dict(value) = dict(value) + 1
        Else
            dict.Add value, 1
        End If
    Next i

    ' 結果を出力
    i = 2
    For Each key In dict.Keys
        ws.Cells(i, 3).Value = key
        ws.Cells(i, 4).Value = dict(key)
        i = i + 1
    Next key
End Sub

    課題2：マスタ参照

    A列のコードに対応する名前をマスタシートから取得してB列に出力するマクロを作成してください。

    解答例 
    Sub MasterLookup()
    Dim wsData As Worksheet, wsMaster As Worksheet
    Dim dict As Object
    Dim lastRow As Long
    Dim i As Long
    Dim code As String

    Set wsData = Worksheets("データ")
    Set wsMaster = Worksheets("マスタ")
    Set dict = CreateObject("Scripting.Dictionary")

    ' マスタをDictionaryに読み込み
    lastRow = wsMaster.Cells(wsMaster.Rows.Count, 1).End(xlUp).Row
    For i = 2 To lastRow
        dict.Add wsMaster.Cells(i, 1).Value, wsMaster.Cells(i, 2).Value
    Next i

    ' データシートを処理
    lastRow = wsData.Cells(wsData.Rows.Count, 1).End(xlUp).Row
    For i = 2 To lastRow
        code = wsData.Cells(i, 1).Value
        If dict.Exists(code) Then
            wsData.Cells(i, 2).Value = dict(code)
        Else
            wsData.Cells(i, 2).Value = "該当なし"
        End If
    Next i
End Sub
