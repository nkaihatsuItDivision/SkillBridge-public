    レッスン8.6: マップモジュール

    🎯 このレッスンの目標

    • ダンジョンマップの構造を実装する
    • プレイヤーの移動処理を作成する
    • マップイベント（敵遭遇、宝箱など）を実装する

    学習時間: 約2時間

    📚 このレッスンで作るもの

    ファイル構成

    src/
├── map.h           ← マップ定義
└── map.c           ← マップ実装

    機能

    1. マップデータ: 複数階層のダンジョン
    2. 移動システム: WASD操作
    3. イベント処理: 敵、宝箱、階段など

    🔧 Step 1: マップ定義（map.h）

    マップの構造

    #ifndef MAP_H
#define MAP_H

#include "player.h"
#include "inventory.h"

// マップサイズ
#define MAP_WIDTH  12
#define MAP_HEIGHT 8
#define MAP_FLOORS 6

// マップチップ
typedef enum {
    TILE_FLOOR = '.',       // 床
    TILE_WALL  = '#',       // 壁
    TILE_PLAYER = '@',      // プレイヤー
    TILE_STAIRS_DOWN = '>',  // 下り階段
    TILE_STAIRS_UP = '<',    // 上り階段
    TILE_ENEMY = 'E',       // 通常敵
    TILE_BOSS = 'B',        // ボス
    TILE_CHEST = 'C',       // 宝箱
    TILE_SHOP = 'S',        // ショップ
    TILE_HEAL = 'H',        // 回復ポイント
    TILE_EXIT = 'G'         // ゴール（クリア）
} TileType;

    🔧 Step 3: マップ初期化と移動

    // マップ初期化
void map_init(MapState *map) {
    if (map == NULL) return;

    // マップデータコピー
    for (int f = 0; f < MAP_FLOORS; f++) {
        for (int y = 0; y < MAP_HEIGHT; y++) {
            strncpy(map->tiles[f][y], g_map_data[f][y], MAP_WIDTH);
            map->tiles[f][y][MAP_WIDTH] = '\0';
        }
    }

    // 訪問済み初期化
    memset(map->visited, 0, sizeof(map->visited));

    // 宝箱状態初期化
    memset(map->chest_opened, 0, sizeof(map->chest_opened));

    // 初期位置（1階の適当な床）
    map->current_floor = 0;
    map->player_pos.x = 1;
    map->player_pos.y = 1;

    // 初期位置を訪問済みに
    map->visited[0][1][1] = 1;
}

// 指定座標のタイルを取得
static char get_tile(const MapState *map, int x, int y) {
    if (x < 0 || x >= MAP_WIDTH || y < 0 || y >= MAP_HEIGHT) {
        return TILE_WALL;
    }
    return map->tiles[map->current_floor][y][x];
}

// 移動可能か判定
static int can_move(const MapState *map, int x, int y) {
    char tile = get_tile(map, x, y);
    return tile != TILE_WALL;
}

// プレイヤー移動
int map_move(MapState *map, Direction dir) {
    if (map == NULL) return 0;

    int new_x = map->player_pos.x;
    int new_y = map->player_pos.y;

    switch (dir) {
        case DIR_UP:    new_y--; break;
        case DIR_DOWN:  new_y++; break;
        case DIR_LEFT:  new_x--; break;
        case DIR_RIGHT: new_x++; break;
    }

    if (!can_move(map, new_x, new_y)) {
        printf("そちらには進めない...\n");
        return 0;
    }

    map->player_pos.x = new_x;
    map->player_pos.y = new_y;

    // 訪問済みにする
    map->visited[map->current_floor][new_y][new_x] = 1;

    return 1;
}

// 現在位置のイベントチェック
MapEvent map_check_event(const MapState *map) {
    if (map == NULL) return EVENT_NONE;

    char tile = get_tile(map, map->player_pos.x, map->player_pos.y);

    switch (tile) {
        case TILE_ENEMY:       return EVENT_ENEMY;
        case TILE_BOSS:        return EVENT_BOSS;
        case TILE_CHEST:       return EVENT_CHEST;
        case TILE_SHOP:        return EVENT_SHOP;
        case TILE_HEAL:        return EVENT_HEAL;
        case TILE_STAIRS_DOWN: return EVENT_STAIRS_DOWN;
        case TILE_STAIRS_UP:   return EVENT_STAIRS_UP;
        case TILE_EXIT:        return EVENT_GOAL;
        default:               return EVENT_NONE;
    }
}

// 階段処理
void map_handle_stairs(MapState *map, int going_down) {
    if (map == NULL) return;

    int new_floor = map->current_floor + (going_down ? 1 : -1);

    if (new_floor < 0) {
        printf("これ以上上には行けない。\n");
        return;
    }
    if (new_floor >= MAP_FLOORS) {
        printf("これ以上下には行けない。\n");
        return;
    }

    map->current_floor = new_floor;
    printf("\n--- %s（%d階）---\n", 
           map_get_floor_name(new_floor), new_floor + 1);

    // 階段の位置を探して移動
    for (int y = 0; y < MAP_HEIGHT; y++) {
        for (int x = 0; x < MAP_WIDTH; x++) {
            char tile = map->tiles[new_floor][y][x];
            // 下りてきた場合は上り階段、上がってきた場合は下り階段を探す
            if ((going_down && tile == TILE_STAIRS_UP) ||
                (!going_down && tile == TILE_STAIRS_DOWN)) {
                map->player_pos.x = x;
                map->player_pos.y = y;
                map->visited[new_floor][y][x] = 1;
                return;
            }
        }
    }

    // 見つからない場合は適当な床に配置
    map->player_pos.x = 1;
    map->player_pos.y = 1;
}

    🔧 Step 4: 宝箱とイベント処理

    // 宝箱オープン
int map_open_chest(MapState *map, Inventory *inv) {
    if (map == NULL || inv == NULL) return 0;

    int floor = map->current_floor;
    int x = map->player_pos.x;
    int y = map->player_pos.y;

    // 宝箱IDを座標から計算（簡易版）
    int chest_id = y * MAP_WIDTH + x;
    if (chest_id >= 10) chest_id = chest_id % 10;

    // 既に開けている
    if (map->chest_opened[floor][chest_id]) {
        printf("この宝箱は既に開けた。\n");
        return 0;
    }

    // 宝箱を開ける
    map->chest_opened[floor][chest_id] = 1;

    // 中身を決定（階層で変化）
    int item_id = 0;
    int gold = 0;

    switch (floor) {
        case 0:  // 1階
            item_id = 1;  // 薬草
            gold = 30;
            break;
        case 1:  // 2階
            item_id = 2;  // 上薬草
            gold = 50;
            break;
        case 2:  // 3階
            item_id = 9;  // 銅の剣
            gold = 80;
            break;
        case 3:  // 4階
            item_id = 3;  // 万能薬
            gold = 120;
            break;
        default:
            gold = 200;
            break;
    }

    printf("宝箱を開けた！\n");

    if (item_id > 0) {
        inventory_add(inv, item_id, 1);
        const Item *item = item_get(item_id);
        if (item) {
            printf("→ %s を手に入れた！\n", item->name);
        }
    }
    if (gold > 0) {
        inv->gold += gold;
        printf("→ %dG を手に入れた！\n", gold);
    }

    // マップから宝箱を消す（床に変える）
    map->tiles[floor][y][x] = TILE_FLOOR;

    return 1;
}

// 階層取得
int map_get_floor(const MapState *map) {
    if (map == NULL) return 0;
    return map->current_floor;
}

    🔧 Step 5: マップ表示

    // マップ表示（全体）
void map_print(const MapState *map) {
    if (map == NULL) return;

    int floor = map->current_floor;

    printf("\n=== %s（%d階）===\n", 
           map_get_floor_name(floor), floor + 1);

    for (int y = 0; y < MAP_HEIGHT; y++) {
        for (int x = 0; x < MAP_WIDTH; x++) {
            if (x == map->player_pos.x && y == map->player_pos.y) {
                printf("%c", TILE_PLAYER);
            } else {
                printf("%c", map->tiles[floor][y][x]);
            }
        }
        printf("\n");
    }

    printf("\n[W:上 A:左 S:下 D:右] [I:アイテム] [Q:メニュー]\n");
}

// ミニマップ表示（訪問済みのみ）
void map_print_minimap(const MapState *map) {
    if (map == NULL) return;

    int floor = map->current_floor;

    printf("\n--- ミニマップ ---\n");

    for (int y = 0; y < MAP_HEIGHT; y++) {
        for (int x = 0; x < MAP_WIDTH; x++) {
            if (x == map->player_pos.x && y == map->player_pos.y) {
                printf("@");
            } else if (map->visited[floor][y][x]) {
                printf("%c", map->tiles[floor][y][x]);
            } else {
                printf(" ");  // 未訪問は非表示
            }
        }
        printf("\n");
    }
}

    🎮 使用例

    メインループでの使用

    #include "map.h"
#include "player.h"
#include "inventory.h"
#include "battle.h"

void game_loop(void) {
    Player player;
    Inventory inv;
    MapState map;

    player_init(&player, "勇者");
    inventory_init(&inv);
    map_init(&map);

    int game_over = 0;

    while (!game_over) {
        map_print(&map);

        printf("> ");
        char cmd = getchar();
        while (getchar() != '\n');  // バッファクリア

        Direction dir = -1;
        switch (cmd) {
            case 'w': case 'W': dir = DIR_UP;    break;
            case 's': case 'S': dir = DIR_DOWN;  break;
            case 'a': case 'A': dir = DIR_LEFT;  break;
            case 'd': case 'D': dir = DIR_RIGHT; break;
            case 'i': case 'I':
                inventory_print(&inv);
                continue;
            case 'q': case 'Q':
                // メニュー処理
                continue;
        }

        if (dir >= 0) {
            if (map_move(&map, dir)) {
                // 移動成功、イベントチェック
                MapEvent event = map_check_event(&map);

                switch (event) {
                    case EVENT_ENEMY:
                        printf("敵が現れた！\n");
                        // battle_start(...);
                        break;
                    case EVENT_BOSS:
                        printf("ボスが現れた！\n");
                        // boss_battle(...);
                        break;
                    case EVENT_CHEST:
                        map_open_chest(&map, &inv);
                        break;
                    case EVENT_HEAL:
                        player.hp = player.max_hp;
                        player.mp = player.max_mp;
                        printf("回復ポイント！HPとMPが全回復した！\n");
                        break;
                    case EVENT_STAIRS_DOWN:
                        printf("下り階段がある。降りる？(y/n) ");
                        if (getchar() == 'y') {
                            map_handle_stairs(&map, 1);
                        }
                        while (getchar() != '\n');
                        break;
                    case EVENT_STAIRS_UP:
                        printf("上り階段がある。上る？(y/n) ");
                        if (getchar() == 'y') {
                            map_handle_stairs(&map, 0);
                        }
                        while (getchar() != '\n');
                        break;
                    case EVENT_GOAL:
                        printf("\n★★★ ダンジョンクリア！★★★\n");
                        game_over = 1;
                        break;
                    default:
                        break;
                }
            }
        }
    }
}
