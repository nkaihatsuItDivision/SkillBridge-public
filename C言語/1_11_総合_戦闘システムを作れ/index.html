    🎮 仕様書：戦闘システムを作れ

    概要

    1体の敵と戦闘するRPGの戦闘システムを作成してください。

    動作例

    +========================================+
|       ダンジョン探索 RPG - Week 1      |
+========================================+

  あなたは勇者です。ダンジョンを探索しています...
  （Enterキーでスタート）

===== 戦闘1 =====
  ゴブリンが現れた！

+----------------------------------------+
|  [ゴブリン]

    📋 必須機能

    1. 基本システム

    • [ ] タイトル画面
    • [ ] プレイヤーステータス（HP, MP, 攻撃力, 防御力, レベル, 経験値）
    • [ ] 敵ステータス（HP, 攻撃力, 防御力, 経験値報酬）
    • [ ] ステータス表示

    2. 戦闘コマンド

    コマンド 効果
    1. 攻撃 通常攻撃（ダメージ = 攻撃力 - 敵防御力、ランダム変動あり）
    2. 防御 防御（このターンの被ダメージ半減）
    3. スキル MP10消費、ダメージ1.5倍
    4. 逃げる 40%の確率で逃走成功

    3. ダメージ計算

    • 基本ダメージ = 攻撃力 - 防御力
    • ±25%のランダム変動
    • 15%の確率でクリティカル（1.5倍）
    • 最低ダメージ = 1

    4. 戦闘ループ

    • プレイヤー → 敵 の順で行動
    • どちらかのHPが0以下で終了
    • 勝利時：経験値獲得、レベルアップ判定
    • 敗北時：ゲームオーバー

    5. レベルアップ

    • 必要経験値 = レベル × 100（Lv1: 100, Lv2: 200...）
    • レベルアップ時：
    • HP +15
    • MP +5
    • 攻撃力 +3
    • 防御力 +2
    • HP/MP全回復

    6. 敵の種類（ランダム出現）

    敵 HP 攻撃力 防御力 経験値
    スライム 40±10 10 3 20
    ゴブリン 60±10 15 5 35
    コウモリ 30±10 12 2 25

// === 戦闘 ===
int calc_damage(int attack, int defense, int *is_crit) {
    *is_crit = 0;

    int base = attack - defense;
    if (base < 1) base = 1;

    int range = base / 4;
    if (range < 1) range = 1;
    int variation = rand() % (range * 2 + 1) - range;
    int damage = base + variation;

    if (rand() % 100 < 15) {
        *is_crit = 1;
        damage = (int)(damage * 1.5);
    }

    if (damage < 1) damage = 1;
    return damage;
}

void do_player_attack(void) {
    int is_crit;
    int damage = calc_damage(player_attack, enemy_defense, &is_crit);

    printf("\n  勇者の攻撃！");
    if (is_crit) printf(" ★★★ 会心の一撃！ ★★★");
    printf("\n  %dダメージ！\n", damage);

    enemy_hp -= damage;
    if (enemy_hp < 0) enemy_hp = 0;
}

void do_player_defend(int *defending) {
    printf("\n  勇者は身を守っている！\n");
    printf("  （このターン、受けるダメージが半減）\n");
    *defending = 1;
}

void do_player_skill(void) {
    if (player_mp < 10) {
        printf("\n  MPが足りません！\n");
        return;
    }

    player_mp -= 10;

    int skill_attack = (int)(player_attack * 1.5);
    int is_crit;
    int damage = calc_damage(skill_attack, enemy_defense, &is_crit);

    printf("\n  勇者のパワースラッシュ！（MP -10）\n");
    if (is_crit) printf("  ★★★ 会心の一撃！ ★★★\n");
    printf("  %dダメージ！\n", damage);

    enemy_hp -= damage;
    if (enemy_hp < 0) enemy_hp = 0;
}

int try_escape(void) {
    printf("\n  勇者は逃げ出そうとした...\n");

    if (rand() % 100 < 40) {
        printf("  うまく逃げ切れた！\n");
        return 1;
    } else {
        printf("  しかし回り込まれてしまった！\n");
        return 0;
    }
}

void do_enemy_turn(int defending) {
    int is_crit;
    int effective_defense = defending ? player_defense * 2 : player_defense;
    int damage = calc_damage(enemy_attack, effective_defense, &is_crit);

    printf("\n  %sの攻撃！", enemy_names[enemy_type]);
    if (defending) printf(" ガード！");
    printf("\n  %dダメージ！\n", damage);

    player_hp -= damage;
    if (player_hp < 0) player_hp = 0;
}

void check_level_up(void) {
    int required = player_level * 100;

    if (player_exp >= required) {
        player_level++;
        player_exp -= required;

        player_max_hp += 15;
        player_max_mp += 5;
        player_attack += 3;
        player_defense += 2;

        player_hp = player_max_hp;
        player_mp = player_max_mp;

        printf("\n  ** レベルアップ！ **\n");
        printf("  レベル%dになった！\n", player_level);
        printf("  HP: %d  MP: %d  攻撃力: %d  防御力: %d\n",
               player_max_hp, player_max_mp, player_attack, player_defense);
    }
}

int main(void) {
    srand(time(NULL));

    init_player();
    show_title();
    getchar();

    int battle_count = 0;
    int game_over = 0;

    while (!game_over) {
        battle_count++;
        printf("\n===== 戦闘%d =====\n", battle_count);

        init_enemy();
        printf("  %sが現れた！\n", enemy_names[enemy_type]);

        int battle_end = 0;

        while (!battle_end) {
            show_status();
            show_menu();

            int cmd;
            scanf("%d", &cmd);

            int is_defending = 0;
            int skip_enemy = 0;

            switch (cmd) {
                case 1:
                    do_player_attack();
                    break;
                case 2:
                    do_player_defend(&is_defending);
                    break;
                case 3:
                    do_player_skill();
                    break;
                case 4:
                    if (try_escape()) {
                        battle_end = 1;
                        skip_enemy = 1;
                    }
                    break;
                default:
                    printf("  無効なコマンドです！\n");
                    continue;
            }

            if (skip_enemy) continue;

            if (enemy_hp <= 0) {
                printf("\n+========================================+\n");
                printf("|         *** 勝利！ ***                 |\n");
                printf("+========================================+\n");

                player_exp += enemy_exp_reward;
                printf("  %dの経験値を獲得！\n", enemy_exp_reward);

                check_level_up();
                battle_end = 1;
                continue;
            }

            do_enemy_turn(is_defending);

            if (player_hp <= 0) {
                printf("\n+========================================+\n");
                printf("|       *** ゲームオーバー ***           |\n");
                printf("+========================================+\n");
                game_over = 1;
                battle_end = 1;
            }
        }

        if (!game_over) {
            printf("\n  続けますか？（1=はい / 0=いいえ）: ");
            int choice;
            scanf("%d", &choice);
            if (choice != 1) {
                game_over = 1;
            }
        }
    }

    printf("\n+========================================+\n");
    printf("|        お疲れさまでした！              |\n");
    printf("|   戦闘回数: %d\n", battle_count);
    printf("|   最終レベル: %d\n", player_level);
    printf("+========================================+\n");

    return 0;
}
