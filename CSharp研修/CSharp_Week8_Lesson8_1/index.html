    1. ダッシュボードとは？

    ダッシュボード = 重要な情報を1画面にまとめて見やすく表示する画面

    業務アプリやWebサービスで必ず見かける画面です。タイピングゲームでは「プレイヤーの成長と実績」を一覧表示します。

    Panel CreateStatCard(string title, string value, Color accentColor)
{
    var card = new Panel
    {
        Size = new Size(140, 90),
        BackColor = Color.White,
        BorderStyle = BorderStyle.FixedSingle
    };

    var lblTitle = new Label
    {
        Text = title,
        Font = new Font("Arial", 9),
        ForeColor = Color.Gray,
        Location = new Point(10, 10),
        AutoSize = true
    };

    var lblValue = new Label
    {
        Text = value,
        Font = new Font("Arial", 22, FontStyle.Bold),
        ForeColor = accentColor,
        Location = new Point(10, 38),
        AutoSize = true
    };

    card.Controls.AddRange(new Control[] { lblTitle, lblValue });
    return card;
}

    折れ線グラフ（スコア推移）

    using ScottPlot;

var formsPlot = new ScottPlot.WinForms.FormsPlot
{
    Size = new Size(550, 250),
    Location = new Point(20, 110)
};

// データ準備
var scores = scoreService.GetPlayerScores(playerName)
    .OrderBy(s => s.PlayDate)
    .ToList();

double[] xs = Enumerable.Range(1, scores.Count).Select(i => (double)i).ToArray();
double[] ys = scores.Select(s => (double)s.ScoreValue).ToArray();

formsPlot.Plot.Add.Scatter(xs, ys);
formsPlot.Plot.Title("スコアの推移");
formsPlot.Plot.XLabel("プレイ回数");
formsPlot.Plot.YLabel("スコア");
formsPlot.Refresh();

this.Controls.Add(formsPlot);

    棒グラフ（難易度別平均）

    var barPlot = new ScottPlot.WinForms.FormsPlot
{
    Size = new Size(250, 200),
    Location = new Point(20, 380)
};

// 難易度別の平均スコア
var byDiff = scores.GroupBy(s => s.Difficulty)
    .Select(g => new { Diff = g.Key, Avg = g.Average(s => s.ScoreValue) })
    .ToList();

double[] values = byDiff.Select(d => d.Avg).ToArray();
double[] positions = Enumerable.Range(0, values.Length).Select(i => (double)i).ToArray();

var bars = barPlot.Plot.Add.Bars(positions, values);
barPlot.Plot.Title("難易度別 平均スコア");
barPlot.Refresh();

this.Controls.Add(barPlot);

    public class DashboardForm : Form
