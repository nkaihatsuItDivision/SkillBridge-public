    総合：タイピングゲームv6を作れ②（モード選択画面・設定画面）

    🎯 このレッスンの目標

    • 複数フォーム間の画面遷移を実装する
    • メイン画面 → モード選択 → ゲーム → 結果の流れを作る
    • タイピングゲームv6 を完成させる

    仕様書：v6②（画面遷移）

    画面構成

    MainForm（メイン画面）
├── ゲーム開始 → ModeSelectForm → GameForm → ResultForm
├── ランキング → RankingForm
├── 統計 → StatsForm
└── 終了

    各画面の役割

    画面 クラス名 内容
    メイン MainForm プレイヤー名入力、メニュー
    モード選択 ModeSelectForm 難易度・モード選択
    ゲーム GameForm タイピング本体（v6①）
    結果 ResultForm スコア・評価・詳細
    ランキング RankingForm TOP10（DataGridView）
    統計 StatsForm プレイヤー統計

    画面遷移の実装

    // MainForm → ModeSelectForm
private void BtnStart_Click(object sender, EventArgs e)
{
    var modeForm = new ModeSelectForm();
    if (modeForm.ShowDialog() == DialogResult.OK)
    {
        var config = modeForm.SelectedConfig;
        var gameForm = new GameForm(config, _playerName);
        gameForm.ShowDialog();

        if (gameForm.GameResult != null)
        {
            var resultForm = new ResultForm(gameForm.GameResult);
            resultForm.ShowDialog();
        }
    }
}

    public class ResultForm : Form
{
    public bool PlayAgain { get; private set; }

    public ResultForm(GameResult result)
    {
        this.Text = "ゲーム結果";
        this.Size = new Size(400, 400);
        this.StartPosition = FormStartPosition.CenterParent;

        string grade = result.Score switch
        {
            >= 800 => "S", >= 600 => "A", >= 400 => "B", >= 200 => "C", _ => "F"
        };

        // スコア表示
        var lblScore = new Label
        {
            Text = $"{result.Score} 点",
            Font = new Font("Arial", 36, FontStyle.Bold),
            ForeColor = grade == "S" ? Color.Gold : Color.DarkBlue,
            AutoSize = true, Location = new Point(120, 50)
        };

        var lblGrade = new Label
        {
            Text = $"{grade} ランク",
            Font = new Font("Arial", 20),
            AutoSize = true, Location = new Point(140, 110)
        };

        // 詳細
        var lblDetails = new Label
        {
            Text = $"WPM：{result.Wpm:F1}\n正確性：{result.Accuracy:F1}%\n" +
                   $"正解：{result.Correct}/{result.Total}\n難易度：{result.Difficulty}",
            Font = new Font("Arial", 12),
            AutoSize = true, Location = new Point(100, 170)
        };

        var btnRetry = new Button { Text = "もう一度", Location = new Point(60, 300), Size = new Size(120, 40) };
        btnRetry.Click += (s, e) => { PlayAgain = true; this.DialogResult = DialogResult.OK; this.Close(); };

        var btnBack = new Button { Text = "メニューへ", Location = new Point(210, 300), Size = new Size(120, 40) };
        btnBack.Click += (s, e) => { this.DialogResult = DialogResult.Cancel; this.Close(); };

        this.Controls.AddRange(new Control[] { lblScore, lblGrade, lblDetails, btnRetry, btnBack });
    }
}

    // ShowDialog: モーダル（親画面を操作できない）
