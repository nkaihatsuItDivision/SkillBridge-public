    🔍 ステップ1: Pageableとは何か

    Pageableは、Spring Data JPAが提供するページネーション機能の中核です。大量のデータを効率的に分割表示するための仕組みです。

    なぜページネーションが重要なのか

    // 悪い例：全データを一度に取得（パフォーマンス問題）
List<Student> allStudents = studentRepository.findAll(); // 10,000件のデータ
// → メモリ大量消費、画面表示の遅延、ユーザビリティ悪化

// 良い例：ページネーション活用（企業システムレベル）
Pageable pageable = PageRequest.of(0, 20); // 1ページ目、20件ずつ
Page<Student> studentPage = studentRepository.findAll(pageable);
// → 効率的なメモリ使用、高速表示、優れたユーザビリティ

public class StudentSearchServiceImpl implements StudentSearchService {

    private StudentRepository studentRepository;

    // Week 10.1で実装済み：基本検索

            throws ServiceException {

    // Week 10.3新規実装：ページネーション対応検索
    public Page<StudentDto> searchStudentsWithPaging(StudentSearchCriteria criteria, 
                                                     Pageable pageable) 
            throws ServiceException {
        try {
            log.info("ページネーション検索開始: criteria={}, page={}, size={}", 
                    criteria, pageable.getPageNumber(), pageable.getPageSize());

            // Week 10.1のSpecification活用
            Specification<Student> spec = StudentSpecifications.searchByCriteria(criteria);

            // ページネーション検索実行
            Page<Student> studentPage = studentRepository.findAll(spec, pageable);

            // Entity → DTO変換（Page対応）
            Page<StudentDto> result = studentPage.map(this::convertToDto);

            log.info("ページネーション検索完了: 総件数={}, 総ページ数={}", 
                    result.getTotalElements(), result.getTotalPages());

            return result;

        } catch (Exception e) {
            log.error("ページネーション検索でエラーが発生しました", e);
            throw new ServiceException("検索処理中にエラーが発生しました", e);
        }
    }

    // 検索結果統計情報取得
    public SearchResultSummary getSearchSummary(StudentSearchCriteria criteria) 
            throws ServiceException {
        try {
            // 基本統計
            long totalCount = studentRepository.count(
                StudentSpecifications.searchByCriteria(criteria));

            // 専攻別統計
            Map<String, Long> majorStats = studentRepository.countByMajorGrouping(criteria);

            // 学年別統計
            Map<Integer, Long> gradeStats = studentRepository.countByGradeGrouping(criteria);

            return new SearchResultSummary(totalCount, majorStats, gradeStats);

        } catch (Exception e) {
            throw new ServiceException("統計情報取得でエラーが発生しました", e);
        }
    }
}

    @Controller
@RequestMapping("/students")
public class StudentSearchController {

    @Autowired
    private StudentSearchService studentSearchService;

    // Week 10.2で実装済み：検索フォーム表示
    @GetMapping("/search")
    public String searchForm(Model model) {
        // 既存実装（変更なし）
    }

    // Week 10.2で実装済み：検索実行（PRGパターン）
    @PostMapping("/search")
    public String executeSearch(@Valid @ModelAttribute("searchCriteria") StudentSearchForm searchForm,
                               BindingResult bindingResult,
                               RedirectAttributes redirectAttributes,
                               Model model) {
        // 既存実装に検索結果画面リダイレクト追加
        if (bindingResult.hasErrors()) {
            return "student/search-form";
        }

        try {
            StudentSearchCriteria criteria = searchForm.toCriteria();

            // 検索条件をFlashAttributeに保存（PRGパターン）
            redirectAttributes.addFlashAttribute("searchCriteria", criteria);
            redirectAttributes.addFlashAttribute("searchForm", searchForm);

            return "redirect:/students/search/results";

        } catch (ServiceException e) {
            bindingResult.reject("search.error", e.getMessage());
            return "student/search-form";
        }
    }

    // Week 10.3新規実装：検索結果表示（ページネーション対応）
    @GetMapping("/search/results")
    public String searchResults(@RequestParam(defaultValue = "0") int page,
                               @RequestParam(defaultValue = "10") int size,
                               @RequestParam(defaultValue = "id") String sortBy,
                               @RequestParam(defaultValue = "asc") String sortDirection,
                               @ModelAttribute("searchCriteria") StudentSearchCriteria criteria,
                               Model model) {

        try {
            // パラメータバリデーション
            if (page < 0) page = 0;
            if (size < 1 || size > 100) size = 10;

            // Pageableオブジェクト作成
            Sort.Direction direction = Sort.Direction.fromString(sortDirection);
            Sort sort = Sort.by(direction, sortBy);
            Pageable pageable = PageRequest.of(page, size, sort);

            // Week 10.1-10.2基盤活用：ページネーション検索実行
            Page<StudentDto> searchResults = studentSearchService.searchStudentsWithPaging(criteria, pageable);

            // 検索結果統計情報取得
            SearchResultSummary summary = studentSearchService.getSearchSummary(criteria);

            // 検索結果・ページ情報をモデルに設定
            model.addAttribute("searchResults", searchResults);
            model.addAttribute("searchSummary", summary);
            model.addAttribute("currentPage", page);
            model.addAttribute("totalPages", searchResults.getTotalPages());
            model.addAttribute("totalElements", searchResults.getTotalElements());
            model.addAttribute("sortBy", sortBy);
            model.addAttribute("sortDirection", sortDirection);
            model.addAttribute("pageSize", size);

            // ページネーション用のページ番号リスト生成
            List<Integer> pageNumbers = generatePageNumbers(page, searchResults.getTotalPages());
            model.addAttribute("pageNumbers", pageNumbers);

            return "student/search-results";

        } catch (ServiceException e) {
            model.addAttribute("errorMessage", "検索結果の表示中にエラーが発生しました: " + e.getMessage());
            return "student/search-form";
        }
    }

    // ページネーション用のページ番号リスト生成
    private List<Integer> generatePageNumbers(int currentPage, int totalPages) {
        List<Integer> pageNumbers = new ArrayList<>();
        int startPage = Math.max(0, currentPage - 2);
        int endPage = Math.min(totalPages - 1, currentPage + 2);

        for (int i = startPage; i <= endPage; i++) {
            pageNumbers.add(i);
        }

        return pageNumbers;
    }
}

    private final String direction;
    private final String displayName;

    SortOption(String sortBy, String direction, String displayName) {
        this.sortBy = sortBy;
        this.direction = direction;
        this.displayName = displayName;
    }

    // getter メソッド
    public String getSortBy() { return sortBy; }
    public String getDirection() { return direction; }
    public String getDisplayName() { return displayName; }

    // Sort オブジェクト生成
    public Sort createSort() {
        Sort.Direction dir = Sort.Direction.fromString(direction);
        return Sort.by(dir, sortBy);
    }

    // 表示用の文字列生成
    public static SortOption fromParameters(String sortBy, String direction) {
        for (SortOption option : values()) {
            if (option.sortBy.equals(sortBy) && option.direction.equals(direction)) {
                return option;
            }
        }
        return NAME_ASC; // デフォルト
    }
}

    // 検索履歴管理
class SearchHistory {
    constructor() {
        this.maxHistory = 5;
        this.storageKey = 'studentSearchHistory';
    }

    // 検索履歴に追加
    addSearch(criteria) {
        const history = this.getHistory();
        const searchItem = {
            criteria: criteria,
            timestamp: new Date().toISOString(),
            displayText: this.generateDisplayText(criteria)
        };

        // 重複チェック
        const existingIndex = history.findIndex(item => 
            JSON.stringify(item.criteria) === JSON.stringify(criteria)
        );

        if (existingIndex >= 0) {
            history.splice(existingIndex, 1);
        }

        history.unshift(searchItem);

        // 最大件数制限
        if (history.length > this.maxHistory) {
            history.splice(this.maxHistory);
        }

        localStorage.setItem(this.storageKey, JSON.stringify(history));
    }

    // 検索履歴取得
    getHistory() {
        const stored = localStorage.getItem(this.storageKey);
        return stored ? JSON.parse(stored) : [];
    }

    // 表示テキスト生成
    generateDisplayText(criteria) {
        const parts = [];
        if (criteria.name) parts.push(`名前:${criteria.name}`);
        if (criteria.grade) parts.push(`学年:${criteria.grade}年`);
        if (criteria.major) parts.push(`専攻:${criteria.major}`);
        if (criteria.email) parts.push(`メール:${criteria.email}`);

        return parts.length > 0 ? parts.join(', ') : '全件検索';
    }

    // 履歴表示HTML生成
    generateHistoryHTML() {
        const history = this.getHistory();
        if (history.length === 0) {
            return '<p class="text-muted">検索履歴はありません</p>';
        }

        return history.map((item, index) => `
            <div class="search-history-item">
                <a href="#" onclick="applySearchHistory(${index})" class="text-decoration-none">
                    <i class="fas fa-history"></i> ${item.displayText}
                </a>
                <small class="text-muted d-block">
                    ${new Date(item.timestamp).toLocaleString()}
                </small>
            </div>
        `).join('');
    }
}

        // フォームに値を設定
