    service パッケージに CsvExportService クラスを作成します：

    package com.studentmanagement.service;

import com.studentmanagement.entity.Student;
import org.springframework.stereotype.Service;
import java.util.List;
import java.time.format.DateTimeFormatter;

@Service
public class CsvExportService {

    private static final DateTimeFormatter DATE_FORMATTER = 
        DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");

    /**
     * 学生リストをCSV形式の文字列に変換
     */
    public String generateStudentListCsv(List<Student> students) {
        StringBuilder csv = new StringBuilder();

        // Section A: ヘッダー行の生成
        csv.append("ID,名前,メールアドレス,専攻,登録日,削除フラグ\n");

        // Section B: データ行の生成
        for (Student student : students) {
            csv.append(student.getId()).append(",")
               .append(escapeCSV(student.getName())).append(",")
               .append(escapeCSV(student.getEmail())).append(",")
               .append(escapeCSV(student.getMajor())).append(",")
               .append(student.getCreatedAt().format(DATE_FORMATTER)).append(",")
               .append(student.isDeleted() ? "削除済み" : "有効").append("\n");
        }

        return csv.toString();
    }

    /**
     * Section C: CSV文字列のエスケープ処理
     */
    private String escapeCSV(String value) {
        if (value == null) {
            return "";
        }

        // カンマ、ダブルクォート、改行が含まれる場合はエスケープが必要
        if (value.contains(",") || value.contains("\"") || value.contains("\n")) {
            // ダブルクォートを２重にしてエスケープ
            return "\"" + value.replace("\"", "\"\"") + "\"";
        }

        return value;
    }

    /**
     * 検索結果をCSVエクスポート（Week 10.1-10.4連携）
     */
    public String generateSearchResultCsv(List<Student> students, String searchCondition) {
        StringBuilder csv = new StringBuilder();

        // 検索条件をコメントとして追加
        csv.append("# 検索条件: ").append(searchCondition).append("\n");
        csv.append("# エクスポート日時: ")
           .append(java.time.LocalDateTime.now().format(DATE_FORMATTER))
           .append("\n");
        csv.append("\n");

        // 通常のCSV生成
        csv.append(generateStudentListCsv(students));

        return csv.toString();
    }

    /**
     * 選択データをCSVエクスポート（Week 10.6連携）
     */
    public String generateSelectedStudentsCsv(List<Student> students) {
        StringBuilder csv = new StringBuilder();

        // 選択データであることを明記
        csv.append("# 選択されたデータのエクスポート\n");
        csv.append("# 選択件数: ").append(students.size()).append("件\n");
        csv.append("# エクスポート日時: ")
           .append(java.time.LocalDateTime.now().format(DATE_FORMATTER))
           .append("\n");
        csv.append("\n");

        // 通常のCSV生成
        csv.append(generateStudentListCsv(students));

        return csv.toString();
    }
}

    - Section C: データ行の生成処理 - Section D: バイト配列変換処理

    service パッケージに ExcelExportService クラスを作成します：

    package com.studentmanagement.service;

import com.studentmanagement.entity.Student;
import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.springframework.stereotype.Service;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.List;
import java.time.format.DateTimeFormatter;

@Service
public class ExcelExportService {

    private static final DateTimeFormatter DATE_FORMATTER = 
        DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");

    /**
     * 学生リストをExcel形式のバイト配列に変換
     */
    public byte[] generateStudentListExcel(List<Student> students) {
        try (Workbook workbook = new XSSFWorkbook()) {

            // Section A: シートの作成
            Sheet sheet = workbook.createSheet("学生一覧");

            // Section B: ヘッダー行の作成とスタイル設定
            createHeaderRow(sheet, workbook);

            // Section C: データ行の作成
            createDataRows(sheet, students);

            // 列幅の自動調整
            autoSizeColumns(sheet);

            // Section D: バイト配列として返す
            return convertToByteArray(workbook);

        } catch (IOException e) {
            throw new RuntimeException("Excel生成に失敗しました", e);
        }
    }

    /**
     * Section B: ヘッダー行の作成
     */
    private void createHeaderRow(Sheet sheet, Workbook workbook) {
        Row headerRow = sheet.createRow(0);

        // ヘッダースタイルの作成
        CellStyle headerStyle = workbook.createCellStyle();
        Font headerFont = workbook.createFont();
        headerFont.setBold(true);
        headerFont.setColor(IndexedColors.WHITE.getIndex());
        headerStyle.setFont(headerFont);
        headerStyle.setFillForegroundColor(IndexedColors.BLUE.getIndex());
        headerStyle.setFillPattern(FillPatternType.SOLID_FOREGROUND);
        headerStyle.setBorderBottom(BorderStyle.THIN);
        headerStyle.setBorderTop(BorderStyle.THIN);
        headerStyle.setBorderRight(BorderStyle.THIN);
        headerStyle.setBorderLeft(BorderStyle.THIN);

        // ヘッダー列の定義
        String[] headers = {"ID", "名前", "メールアドレス", "専攻", "登録日", "状態"};

        for (int i = 0; i < headers.length; i++) {
            Cell cell = headerRow.createCell(i);
            cell.setCellValue(headers[i]);
            cell.setCellStyle(headerStyle);
        }
    }

    /**
     * Section C: データ行の作成
     */
    private void createDataRows(Sheet sheet, List<Student> students) {
        int rowNum = 1; // ヘッダー行の次から開始

        for (Student student : students) {
            Row row = sheet.createRow(rowNum++);

            // データの設定
            row.createCell(0).setCellValue(student.getId());
            row.createCell(1).setCellValue(student.getName());
            row.createCell(2).setCellValue(student.getEmail());
            row.createCell(3).setCellValue(student.getMajor());
            row.createCell(4).setCellValue(
                student.getCreatedAt().format(DATE_FORMATTER));
            row.createCell(5).setCellValue(
                student.isDeleted() ? "削除済み" : "有効");
        }
    }

    /**
     * 列幅の自動調整
     */
    private void autoSizeColumns(Sheet sheet) {
        for (int i = 0; i < 6; i++) {
            sheet.autoSizeColumn(i);
        }
    }

    /**
     * Section D: Workbookをバイト配列に変換
     */
    private byte[] convertToByteArray(Workbook workbook) throws IOException {
        ByteArrayOutputStream outputStream = new ByteArrayOutputStream();
        workbook.write(outputStream);
        return outputStream.toByteArray();
    }

    /**
     * 検索結果をExcelエクスポート（Week 10.1-10.4連携）
     */
    public byte[] generateSearchResultExcel(List<Student> students, String searchCondition) {
        try (Workbook workbook = new XSSFWorkbook()) {

            // 検索結果シートの作成
            Sheet sheet = workbook.createSheet("検索結果");

            // 検索条件を最初の行に記載
            Row conditionRow = sheet.createRow(0);
            conditionRow.createCell(0).setCellValue("検索条件: " + searchCondition);

            Row dateRow = sheet.createRow(1);
            dateRow.createCell(0).setCellValue("エクスポート日時: " + 
                java.time.LocalDateTime.now().format(DATE_FORMATTER));

            // 空行
            sheet.createRow(2);

            // ヘッダー行（3行目から開始）
            createHeaderRowAtIndex(sheet, workbook, 3);

            // データ行の作成（4行目から開始）
            createDataRowsFromIndex(sheet, students, 4);

            // 列幅の自動調整
            autoSizeColumns(sheet);

            return convertToByteArray(workbook);

        } catch (IOException e) {
            throw new RuntimeException("検索結果Excel生成に失敗しました", e);
        }
    }

    /**
     * 指定行からヘッダーを作成
     */
    private void createHeaderRowAtIndex(Sheet sheet, Workbook workbook, int rowIndex) {
        Row headerRow = sheet.createRow(rowIndex);

        // ヘッダースタイル（前述と同じ）
        CellStyle headerStyle = workbook.createCellStyle();
        Font headerFont = workbook.createFont();
        headerFont.setBold(true);
        headerFont.setColor(IndexedColors.WHITE.getIndex());
        headerStyle.setFont(headerFont);
        headerStyle.setFillForegroundColor(IndexedColors.BLUE.getIndex());
        headerStyle.setFillPattern(FillPatternType.SOLID_FOREGROUND);

        String[] headers = {"ID", "名前", "メールアドレス", "専攻", "登録日", "状態"};

        for (int i = 0; i < headers.length; i++) {
            Cell cell = headerRow.createCell(i);
            cell.setCellValue(headers[i]);
            cell.setCellStyle(headerStyle);
        }
    }

    /**
     * 指定行からデータ行を作成
     */
    private void createDataRowsFromIndex(Sheet sheet, List<Student> students, int startRowIndex) {
        int rowNum = startRowIndex;

        for (Student student : students) {
            Row row = sheet.createRow(rowNum++);

            row.createCell(0).setCellValue(student.getId());
            row.createCell(1).setCellValue(student.getName());
            row.createCell(2).setCellValue(student.getEmail());
            row.createCell(3).setCellValue(student.getMajor());
            row.createCell(4).setCellValue(
                student.getCreatedAt().format(DATE_FORMATTER));
            row.createCell(5).setCellValue(
                student.isDeleted() ? "削除済み" : "有効");
        }
    }
}

    コードの理解ポイント： - XSSFWorkbook: .xlsx形式のExcelファイルを生成 - CellStyle: セルの見た目（背景色、フォント、枠線など）の設定 - autoSizeColumn(): 内容に応じた列幅の自動調整 - ByteArrayOutputStream: ファイルをメモリ上でバイト配列として生成
