    🔍 Controller肥大化による問題

    1. 単一責任原則の違反
    2. Controllerが「画面制御」と「ビジネスロジック」の両方を担当

    3. 1つのクラスが複数の責任を持つため、変更時の影響範囲が拡大

    4. テストの困難性

    5. Webリクエスト/レスポンスと一緒にビジネスロジックもテストする必要

    6. 単体テストが複雑になり、テストケースの作成・保守が困難

    7. 再利用性の低下

    8. ビジネスロジックがController内にあるため、他の画面やAPIで再利用不可

    9. 同じ処理を複数のControllerで重複実装するリスク

    10. 保守性の悪化

    11. ビジネスルール変更時に複数のControllerを修正する必要
    12. コードの可読性が低下し、新しいメンバーの理解が困難

    🔧 @Autowiredによる依存性注入

    Spring Bootの依存性注入（Dependency Injection）により、オブジェクト間の結合度を下げることができます：

    // インターフェースによる抽象化
public interface StudentService {

// 実装クラス
@Service
public class StudentServiceImpl implements StudentService {

    @Autowired
    private StudentRepository studentRepository;

    @Override
    public void registerStudent(StudentData studentData) throws ServiceException {
        try {
            // ビジネスルール1: メールアドレス重複チェック
            validateEmailUnique(studentData.getEmail());

            // ビジネスルール2: 学年と専攻の整合性チェック
            validateGradeAndMajor(studentData.getGrade(), studentData.getMajor());

            // Entity変換とDB保存
            Student student = studentData.toEntity();
            studentRepository.save(student);

        } catch (Exception e) {
            throw new ServiceException("学生登録処理でエラーが発生しました", e);
        }
    }

    private void validateEmailUnique(String email) throws ServiceException {
        if (studentRepository.existsByEmail(email)) {
            throw new ServiceException("このメールアドレスは既に使用されています");
        }
    }

    private void validateGradeAndMajor(int grade, String major) throws ServiceException {
        if (grade == 1 && !Arrays.asList("情報工学", "基礎教養").contains(major)) {
            throw new ServiceException("1年生は情報工学または基礎教養のみ選択可能です");
        }
    }
}

    4. シングルトン管理: デフォルトでServiceクラスはシングルトンとして管理

    @Service
public class StudentServiceImpl implements StudentService {

    @Autowired
    private StudentRepository studentRepository;

    @Autowired
    private StudentHistoryRepository studentHistoryRepository;

    @Transactional
    public void updateStudentWithHistory(Long id, StudentData studentData) throws ServiceException {
        try {
            // 1. 既存データの取得
            Student existingStudent = studentRepository.findById(id)
                .orElseThrow(() -> new ServiceException("学生が見つかりません"));

            // 2. 履歴テーブルへの保存
            StudentHistory history = createHistory(existingStudent);

            // 3. 学生情報の更新
            Student updatedStudent = studentData.toEntity();
            updatedStudent.setId(id);
            studentRepository.save(updatedStudent);

            // 4. ログテーブルの更新
            updateAuditLog(id, "UPDATE", "学生情報更新");

            // すべて成功した場合のみコミット
            // 途中で例外が発生した場合は自動ロールバック

        } catch (Exception e) {
            // @Transactionalにより自動的にロールバック
            throw new ServiceException("学生情報更新に失敗しました", e);
        }
    }

    @Transactional
    public void registerStudent(StudentData studentData) throws ServiceException {
        try {
            // バリデーション（ビジネスルール）
            validateStudentData(studentData);

            // 1. 学生情報の保存
            Student student = studentData.toEntity();
            Student savedStudent = studentRepository.save(student);

            // 2. 初期設定データの作成
            createInitialSettings(savedStudent.getId());

            // 3. 監査ログの記録
            createAuditLog(savedStudent.getId(), "CREATE", "新規学生登録");

        } catch (Exception e) {
            throw new ServiceException("学生登録処理でエラーが発生しました", e);
        }
    }
}

            updatedStudent.setId(id);
            studentRepository.save(updatedStudent);

        } catch (DataNotFoundException | ValidationException e) {
            throw e;
        } catch (Exception e) {
            throw new ServiceException("学生情報更新でエラーが発生しました", e);
        }
    }

    public List<Student> getAllStudents() {
        return studentRepository.findAll();
    }

    public Student getStudentById(Long id) throws ServiceException {
        return studentRepository.findById(id)
            .orElseThrow(() -> new DataNotFoundException("指定された学生が見つかりません"));
    }

    private void validateBusinessRules(StudentData studentData) throws ValidationException {
        // TODO: メールアドレス重複チェック
        if (studentRepository.existsByEmail(studentData.getEmail())) {
            throw new ValidationException("このメールアドレスは既に使用されています");
        }

        // TODO: 学年と専攻の整合性チェック
        if (studentData.getGrade() == 1 && 
            !Arrays.asList("情報工学", "基礎教養").contains(studentData.getMajor())) {
            throw new ValidationException("1年生は情報工学または基礎教養のみ選択可能です");
        }

        // TODO: 年齢制限チェック
        if (studentData.getAge() < 16 || studentData.getAge() > 65) {
            throw new ValidationException("年齢は16歳以上65歳以下で入力してください");
        }
    }
}

    @Controller
public class StudentController {

    @Autowired
    private StudentService studentService;  // RepositoryからServiceに変更

    @PostMapping("/students")
    public String registerStudent(@Valid @ModelAttribute StudentData studentData,
                                  BindingResult bindingResult,
                                  RedirectAttributes redirectAttributes) {
        if (bindingResult.hasErrors()) {
            return "students/register";
        }

        try {
            // TODO: ビジネスロジックはServiceに完全委譲
            studentService.registerStudent(studentData);

            redirectAttributes.addFlashAttribute("successMessage", "学生を登録しました");
            return "redirect:/students";

        } catch (ServiceException e) {
            // TODO: Service例外をキャッチして適切なメッセージを設定
            redirectAttributes.addFlashAttribute("errorMessage", e.getMessage());
            return "redirect:/students/register";
        }
    }

    @PostMapping("/students/{id}")
    public String updateStudent(@PathVariable Long id,
                                @Valid @ModelAttribute StudentData studentData,
                                BindingResult bindingResult,
                                RedirectAttributes redirectAttributes) {
        if (bindingResult.hasErrors()) {
            return "students/edit";
        }

        try {
            studentService.updateStudent(id, studentData);

            redirectAttributes.addFlashAttribute("successMessage", "学生情報を更新しました");
            return "redirect:/students";

        } catch (ServiceException e) {
            redirectAttributes.addFlashAttribute("errorMessage", e.getMessage());
            return "redirect:/students/" + id + "/edit";
        }
    }

    @GetMapping("/students")
    public String listStudents(Model model) {
        // TODO: Service層から学生一覧を取得
        List<Student> students = studentService.getAllStudents();
        model.addAttribute("students", students);
        return "students/list";
    }

    @GetMapping("/students/{id}/edit")
    public String editStudent(@PathVariable Long id, Model model, RedirectAttributes redirectAttributes) {
        try {
            Student student = studentService.getStudentById(id);
            StudentData studentData = StudentData.fromEntity(student);
            model.addAttribute("studentData", studentData);
            return "students/edit";

        } catch (ServiceException e) {
            redirectAttributes.addFlashAttribute("errorMessage", e.getMessage());
            return "redirect:/students";
        }
    }
}
    **リファクタリングのポイント:** - `@Autowired` を `StudentRepository` から `StudentService` に変更 - ビジネスロジックはすべてService層に委譲 - 例外処理は `ServiceException` をキャッチ - Controllerは画面制御のみに専念
