    🎯 このレッスンの目標

    • Excelの数式をPythonから設定できる

    📊 数式の設定

    Excelの数式は文字列としてそのまま設定します。

    import openpyxl

wb = openpyxl.Workbook()
ws = wb.active

ws["A1"] = "項目"
ws["B1"] = "金額"
ws["A2"] = "食費"
ws["B2"] = 3500
ws["A3"] = "交通費"
ws["B3"] = 680
ws["A4"] = "娯楽"
ws["B4"] = 1490

# 数式を文字列で設定
ws["A6"] = "合計"
ws["B6"] = "=SUM(B2:B4)"

ws["A7"] = "平均"
ws["B7"] = "=AVERAGE(B2:B4)"

ws["A8"] = "最大"
ws["B8"] = "=MAX(B2:B4)"

ws["A9"] = "件数"
ws["B9"] = "=COUNTA(B2:B4)"

wb.save("formula_sample.xlsx")

    💡 Pythonで設定した数式はExcelで開くと計算結果が表示されます。

    動的に数式を生成

    # データ行数に応じて数式を自動生成
last_row = ws.max_row
ws.cell(row=last_row + 2, column=2, value=f"=SUM(B2:B{last_row})")

    🔄 複数ファイルの一括処理

    実践例：月別Excelの一括集計

    import openpyxl
import glob
import os

def summarize_all_files(folder):
    """フォルダ内の全Excelファイルを集計する"""
    xlsx_files = sorted(glob.glob(f"{folder}/*.xlsx"))
    summaries = []

    for filepath in xlsx_files:
        try:
            wb = openpyxl.load_workbook(filepath)
            ws = wb.active
            total = 0
            count = 0
            for row in ws.iter_rows(min_row=2, values_only=True):
                if row and row[-1] is not None:
                    try:
                        total += int(row[-1])
                        count += 1
                    except (ValueError, TypeError):
                        pass
            filename = os.path.basename(filepath)
            summaries.append({"file": filename, "count": count, "total": total})
            print(f"  ✅ {filename}: {count}件, {total:,}円")
        except Exception as e:
            print(f"  ⚠️ {filepath}: エラー ({e})")

    return summaries

    🛠️ 実践課題

    ファイル名: practice5_4.py

    3つの月別Excelを作成し、一括集計してレポートExcelを出力してください。

    仕様： 1. monthly_data/ フォルダに1〜3月のサンプルExcelを作成 2. 全ファイルを読み込んで集計 3. 集計結果を書式付きで summary_report.xlsx に出力

    ✅ 解答例 
    import openpyxl
import glob
import os
from openpyxl.styles import Font, PatternFill, Border, Side

# 1. サンプルデータ作成
monthly_data = {
    "data_1月.xlsx": [("1/5", "食費", 3500), ("1/10", "交通費", 680), ("1/20", "娯楽", 1490)],
    "data_2月.xlsx": [("2/3", "食費", 1200), ("2/14", "娯楽", 2500), ("2/28", "通信費", 8500)],
    "data_3月.xlsx": [("3/1", "食費", 950), ("3/15", "日用品", 1800), ("3/25", "食費", 2800)],
}

os.makedirs("monthly_data", exist_ok=True)
for filename, data in monthly_data.items():
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.append(["日付", "カテゴリ", "金額"])
    for row in data:
        ws.append(row)
    wb.save(f"monthly_data/{filename}")

# 2. 一括集計
summaries = []
for fp in sorted(glob.glob("monthly_data/data_*.xlsx")):
    wb = openpyxl.load_workbook(fp)
    ws = wb.active
    total = sum(r[2] for r in ws.iter_rows(min_row=2, values_only=True) if r[2])
    name = os.path.basename(fp).replace("data_", "").replace(".xlsx", "")
    summaries.append({"month": name, "count": ws.max_row - 1, "total": total})

# 3. レポート出力
wb = openpyxl.Workbook()
ws = wb.active
ws.title = "月別集計"

header_fill = PatternFill(start_color="4472C4", end_color="4472C4", fill_type="solid")
border = Border(left=Side("thin"), right=Side("thin"), top=Side("thin"), bottom=Side("thin"))

for col, h in enumerate(["月", "件数", "合計"], start=1):
    c = ws.cell(row=1, column=col, value=h)
    c.font = Font(bold=True, color="FFFFFF")
    c.fill = header_fill
    c.border = border

for i, s in enumerate(summaries, start=2):
    ws.cell(row=i, column=1, value=s["month"]).border = border
    ws.cell(row=i, column=2, value=s["count"]).border = border
    c = ws.cell(row=i, column=3, value=s["total"])
    c.border = border
    c.number_format = "#,##0"

# 合計行
total_row = len(summaries) + 2
ws.cell(row=total_row, column=1, value="合計").font = Font(bold=True)
ws.cell(row=total_row, column=2, value=sum(s["count"] for s in summaries)).font = Font(bold=True)
tc = ws.cell(row=total_row, column=3, value=sum(s["total"] for s in summaries))
tc.font = Font(bold=True)
tc.number_format = "#,##0"

ws.column_dimensions["A"].width = 10
ws.column_dimensions["C"].width = 14

wb.save("summary_report.xlsx")
print("✅ 集計レポート作成完了")
