    学習目標

    • リフレクションの基本概念と実用的な活用方法を理解する

    • フレームワーク開発で必要となる高度な動的プログラミング技法を身につける

    はじめに - リフレクションとは

    リフレクション（Reflection）は、実行時にプログラム自身の構造を調べ、操作する機能です。まさに「鏡で自分を見る」ように、プログラムが自分自身の型情報やメンバーを調べることができます。

    なぜリフレクションが重要なのか

    • フレームワーク開発: ORMやDIコンテナの基盤技術
    • 設定主導開発: XMLやJSONから動的にオブジェクト生成
    • 汎用的なツール: シリアライゼーション、バリデーション
    • デバッグ・診断: 実行時の状態調査
    • プラグインアーキテクチャ: 動的なモジュール読み込み

    メンバー情報の詳細探索

    public class MemberInformationAnalyzer
{
    public void AnalyzeClassMembers()
    {
        Type employeeType = typeof(Employee);

        Console.WriteLine("=== メンバー情報の詳細分析 ===");

        // プロパティ情報の詳細分析
        AnalyzeProperties(employeeType);

        // メソッド情報の詳細分析
        AnalyzeMethods(employeeType);

        // フィールド情報の分析
        AnalyzeFields(employeeType);

        // コンストラクター情報
        AnalyzeConstructors(employeeType);

        // イベント情報（もしあれば）
        AnalyzeEvents(employeeType);
    }

    private void AnalyzeProperties(Type type)
    {
        Console.WriteLine("\n--- プロパティ詳細 ---");

        var properties = type.GetProperties(BindingFlags.Public | BindingFlags.Instance);
        foreach (var property in properties)
        {
            Console.WriteLine($"プロパティ: {property.PropertyType.Name} {property.Name}");
            Console.WriteLine($"  読み取り: {property.CanRead}");
            Console.WriteLine($"  書き込み: {property.CanWrite}");

            // getter/setterの詳細
            if (property.CanRead)
            {
                var getter = property.GetGetMethod();
                Console.WriteLine($"  Getter: {getter?.Name} (仮想: {getter?.IsVirtual}, 静的: {getter?.IsStatic})");
            }

            if (property.CanWrite)
            {
                var setter = property.GetSetMethod();
                Console.WriteLine($"  Setter: {setter?.Name} (仮想: {setter?.IsVirtual}, 静的: {setter?.IsStatic})");
            }

            // カスタム属性の確認
            var attributes = property.GetCustomAttributes();
            if (attributes.Any())
            {
                Console.WriteLine("  属性:");
                foreach (var attr in attributes)
                {
                    Console.WriteLine($"    - {attr.GetType().Name}");
                }
            }
            Console.WriteLine();
        }
    }

    private void AnalyzeMethods(Type type)
    {
        Console.WriteLine("--- メソッド詳細 ---");

        var methods = type.GetMethods(BindingFlags.Public | BindingFlags.Instance | BindingFlags.DeclaredOnly);
        foreach (var method in methods)
        {
            // プロパティのget/setメソッドはスキップ
            if (method.IsSpecialName) continue;

            var parameters = method.GetParameters();
            var parameterString = string.Join(", ", 
                parameters.Select(p => $"{p.ParameterType.Name} {p.Name}"));

            Console.WriteLine($"メソッド: {method.ReturnType.Name} {method.Name}({parameterString})");
            Console.WriteLine($"  仮想: {method.IsVirtual}");
            Console.WriteLine($"  静的: {method.IsStatic}");
            Console.WriteLine($"  抽象: {method.IsAbstract}");
            Console.WriteLine($"  オーバーライド: {method.IsVirtual && method.DeclaringType != method.GetBaseDefinition().DeclaringType}");

            // パラメーター詳細
            if (parameters.Length > 0)
            {
                Console.WriteLine("  パラメーター:");
                foreach (var param in parameters)
                {
                    Console.WriteLine($"    {param.ParameterType.Name} {param.Name}" +
                                    (param.HasDefaultValue ? $" = {param.DefaultValue}" : ""));
                }
            }
            Console.WriteLine();
        }
    }

    private void AnalyzeFields(Type type)
    {
        Console.WriteLine("--- フィールド詳細 ---");

        var fields = type.GetFields(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance);
        foreach (var field in fields)
        {
            Console.WriteLine($"フィールド: {field.FieldType.Name} {field.Name}");
            Console.WriteLine($"  アクセス修飾子: {GetAccessModifier(field)}");
            Console.WriteLine($"  読み取り専用: {field.IsInitOnly}");
            Console.WriteLine($"  リテラル: {field.IsLiteral}");
            Console.WriteLine($"  静的: {field.IsStatic}");
            Console.WriteLine();
        }
    }

    private void AnalyzeConstructors(Type type)
    {
        Console.WriteLine("--- コンストラクター詳細 ---");

        var constructors = type.GetConstructors();
        foreach (var constructor in constructors)
        {
            var parameters = constructor.GetParameters();
            var parameterString = string.Join(", ", 
                parameters.Select(p => $"{p.ParameterType.Name} {p.Name}"));

            Console.WriteLine($"コンストラクター: {constructor.Name}({parameterString})");
            Console.WriteLine($"  アクセス修飾子: {GetAccessModifier(constructor)}");
            Console.WriteLine();
        }
    }

    private void AnalyzeEvents(Type type)
    {
        var events = type.GetEvents();
        if (events.Length > 0)
        {
            Console.WriteLine("--- イベント ---");
            foreach (var eventInfo in events)
            {
                Console.WriteLine($"イベント: {eventInfo.EventHandlerType?.Name} {eventInfo.Name}");
            }
        }
    }

    private string GetAccessModifier(FieldInfo field)
    {
        if (field.IsPublic) return "public";
        if (field.IsPrivate) return "private";
        if (field.IsFamily) return "protected";
        if (field.IsAssembly) return "internal";
        if (field.IsFamilyOrAssembly) return "protected internal";
        return "unknown";
    }

    private string GetAccessModifier(ConstructorInfo constructor)
    {
        if (constructor.IsPublic) return "public";
        if (constructor.IsPrivate) return "private";
        if (constructor.IsFamily) return "protected";
        if (constructor.IsAssembly) return "internal";
        return "unknown";
    }
}

    using System;
using System.Linq.Expressions;
using System.Reflection;
using System.Collections.Concurrent;

public class DynamicObjectFactory
{
    // パフォーマンス向上のためのキャッシュ
    private static readonly ConcurrentDictionary<Type, Func<object[], object>> _constructorCache = new();
    private static readonly ConcurrentDictionary<string, Func<object>> _parameterlessCache = new();

    public void DemonstrateDynamicCreation()
    {
        Console.WriteLine("=== 動的オブジェクト作成のデモンストレーション ===");

        // 方法1: Activator.CreateInstance（簡単だが低速）
        Console.WriteLine("\n1. Activator.CreateInstance:");
        CreateWithActivator();

        // 方法2: コンパイル済み式（高速）
        Console.WriteLine("\n2. コンパイル済み式:");
        CreateWithCompiledExpression();

        // 方法3: ジェネリック制約の活用
        Console.WriteLine("\n3. ジェネリック制約:");
        CreateWithGenericConstraint();

        // 方法4: FormatterServices（特殊用途）
        Console.WriteLine("\n4. FormatterServices:");
        CreateWithFormatterServices();
    }

    private void CreateWithActivator()
    {
        Type employeeType = typeof(Employee);

        // パラメーターなしコンストラクター（存在しない場合はエラー）
        try
        {
            var emp1 = (Employee)Activator.CreateInstance(employeeType);
            Console.WriteLine($"作成失敗予定: {emp1?.Name}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Expected error: {ex.Message}");
        }

        // パラメーター付きコンストラクター
        var emp2 = (Employee)Activator.CreateInstance(employeeType, "田中太郎", "開発部");
        Console.WriteLine($"作成成功: {emp2.Name} - {emp2.Department}");

        // ジェネリック版（型制約がある場合のみ使用可能）
        var stringInstance = Activator.CreateInstance<string>("Hello".ToCharArray());
        Console.WriteLine($"String作成: {stringInstance}");
    }

    private void CreateWithCompiledExpression()
    {
        // より高速な方法：コンパイル済み式
        var constructor = typeof(Employee).GetConstructor(new[] { typeof(string), typeof(string) });
        var createFunc = GetOrCreateConstructorFunction(constructor);

        var emp = (Employee)createFunc(new object[] { "佐藤花子", "営業部" });
        Console.WriteLine($"高速作成: {emp.Name} - {emp.Department}");

        // パフォーマンステスト
        PerformanceComparison();
    }

    private void CreateWithGenericConstraint()
    {
        // ジェネリック制約 where T : new() を使用
        var list = CreateGenericInstance<List<string>>();
        list.Add("テスト");
        Console.WriteLine($"Generic List作成: {list.Count}個の要素");

        var dict = CreateGenericInstance<Dictionary<string, int>>();
        dict["key"] = 42;
        Console.WriteLine($"Generic Dictionary作成: {dict.Count}個の要素");
    }

    private void CreateWithFormatterServices()
    {
        // 注意：コンストラクターを呼ばずにインスタンス作成（特殊用途）
        var uninitializedEmployee = (Employee)System.Runtime.Serialization.FormatterServices
            .GetUninitializedObject(typeof(Employee));

        Console.WriteLine($"未初期化オブジェクト: Name={uninitializedEmployee.Name}, Dept={uninitializedEmployee.Department}");

        // プロパティを手動で設定
        uninitializedEmployee.Name = "手動設定";
        uninitializedEmployee.Department = "特殊部門";
        Console.WriteLine($"手動設定後: {uninitializedEmployee.Name} - {uninitializedEmployee.Department}");
    }

    // 汎用的なコンストラクター関数の生成とキャッシュ
    private Func<object[], object> GetOrCreateConstructorFunction(ConstructorInfo constructor)
    {
        var type = constructor.DeclaringType;

        return _constructorCache.GetOrAdd(type, _ =>
        {
            var parametersParam = Expression.Parameter(typeof(object[]), "parameters");
            var parameterTypes = constructor.GetParameters();
            var argumentExpressions = new Expression[parameterTypes.Length];

            for (int i = 0; i < parameterTypes.Length; i++)
            {
                var parameterType = parameterTypes[i].ParameterType;
                var arrayAccess = Expression.ArrayIndex(parametersParam, Expression.Constant(i));
                argumentExpressions[i] = Expression.Convert(arrayAccess, parameterType);
            }

            var newExpression = Expression.New(constructor, argumentExpressions);
            var lambda = Expression.Lambda<Func<object[], object>>(newExpression, parametersParam);

            return lambda.Compile();
        });
    }

    // ジェネリック制約を使用した型安全な作成
    private T CreateGenericInstance<T>() where T : new()
    {
        return new T();
    }

    private void PerformanceComparison()
    {
        const int iterations = 100000;
        var sw = System.Diagnostics.Stopwatch.StartNew();

        // Activator.CreateInstance
        sw.Restart();
        for (int i = 0; i < iterations; i++)
        {
            var emp = (Employee)Activator.CreateInstance(typeof(Employee), "Test", "Dept");
        }
        sw.Stop();
        Console.WriteLine($"Activator: {sw.ElapsedMilliseconds}ms");

        // Compiled Expression
        var constructor = typeof(Employee).GetConstructor(new[] { typeof(string), typeof(string) });
        var compiledFunc = GetOrCreateConstructorFunction(constructor);

        sw.Restart();
        for (int i = 0; i < iterations; i++)
        {
            var emp = (Employee)compiledFunc(new object[] { "Test", "Dept" });
        }
        sw.Stop();
        Console.WriteLine($"Compiled Expression: {sw.ElapsedMilliseconds}ms");
    }
}

    private void TrySetPropertyValue(object obj, string propertyName, object value)
    {
        var property = GetPropertyInfo(obj.GetType(), propertyName);
        if (property != null && property.CanWrite)
        {
            try
            {

    {

        var propertyAccess = Expression.Property(objParam, property);
        var lambda = Expression.Lambda<Func<T, TProperty>>(propertyAccess, objParam);
        return lambda.Compile();
    }

    private Action<T, TProperty> CreatePropertySetter<T, TProperty>(PropertyInfo property)
    {
        var objParam = Expression.Parameter(typeof(T), "obj");
        var valueParam = Expression.Parameter(typeof(TProperty), "value");
        var propertyAccess = Expression.Property(objParam, property);
        var assignment = Expression.Assign(propertyAccess, valueParam);
        var lambda = Expression.Lambda<Action<T, TProperty>>(assignment, objParam, valueParam);
        return lambda.Compile();
    }

    public class AdvancedValidator
{
    private readonly Dictionary<Type, List<PropertyValidationInfo>> _validationCache = new();

    public ValidationResult ValidateObject(object obj)
    {
        if (obj == null) throw new ArgumentNullException(nameof(obj));

        var result = new ValidationResult();
        var type = obj.GetType();

        // バリデーション情報をキャッシュから取得または生成
        var validationInfos = GetValidationInfo(type);

        // 順序付きでバリデーションを実行
        foreach (var info in validationInfos.OrderBy(v => v.Order))
        {
            ValidateProperty(obj, info, result);
        }

        return result;
    }

    private List<PropertyValidationInfo> GetValidationInfo(Type type)
    {
        if (_validationCache.TryGetValue(type, out var cachedInfo))
        {
            return cachedInfo;
        }

        var validationInfos = new List<PropertyValidationInfo>();

        foreach (var property in type.GetProperties())
        {
            var validationAttributes = property.GetCustomAttributes<ValidationAttribute>().ToList();
            if (validationAttributes.Any())
            {
                validationInfos.Add(new PropertyValidationInfo
                {
                    Property = property,
                    ValidationAttributes = validationAttributes,
                    Order = validationAttributes.Min(a => a.Order)
                });
            }
        }

        _validationCache[type] = validationInfos;
        return validationInfos;
    }

    private void ValidateProperty(object obj, PropertyValidationInfo info, ValidationResult result)
    {
        var value = info.Property.GetValue(obj);

        foreach (var attribute in info.ValidationAttributes.OrderBy(a => a.Order))
        {
            var validationResult = attribute.Validate(value, info.Property.Name);
            result.Merge(validationResult);

            // 必須チェックでエラーがあった場合、他のバリデーションはスキップ
            if (attribute.IsRequired && validationResult.HasErrors)
            {
                break;
            }
        }
    }

    // 条件付きバリデーション
    public ValidationResult ValidateObjectConditional(object obj, Func<PropertyInfo, bool> propertyFilter)
    {
        if (obj == null) throw new ArgumentNullException(nameof(obj));

        var result = new ValidationResult();
        var type = obj.GetType();

        var validationInfos = GetValidationInfo(type)
            .Where(info => propertyFilter(info.Property))
            .OrderBy(v => v.Order);

        foreach (var info in validationInfos)
        {
            ValidateProperty(obj, info, result);
        }

        return result;
    }

    // 差分バリデーション（変更されたプロパティのみ）
    public ValidationResult ValidateChanges(object original, object modified)
    {
        if (original == null || modified == null) 
            throw new ArgumentNullException();

        if (original.GetType() != modified.GetType())
            throw new ArgumentException("オブジェクトの型が一致しません");

        return ValidateObjectConditional(modified, property =>
        {
            var originalValue = property.GetValue(original);
            var modifiedValue = property.GetValue(modified);
            return !Equals(originalValue, modifiedValue);
        });
    }
}

        }

    }

    {

        var type = typeof(T);

        foreach (var property in type.GetProperties())
        {
            var configAttr = property.GetCustomAttribute<ConfigurationAttribute>();
            if (configAttr != null)

            }

    private void LoadPropertyValue(object instance, PropertyInfo property, ConfigurationAttribute configAttr)
    {
        var key = configAttr.Key;

        if (_configurationData.TryGetValue(key, out var configValue))
        {
            try
            {
                var convertedValue = ConvertValue(configValue, property.PropertyType);
                property.SetValue(instance, convertedValue);
                Console.WriteLine($"設定読み込み: {key} = {convertedValue}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"設定変換エラー {key}: {ex.Message}");
                SetDefaultValue(instance, property, configAttr);
            }
        }
        else if (configAttr.IsRequired)
        {
            throw new InvalidOperationException($"必須設定項目が見つかりません: {key}");
        }
        else
        {
            SetDefaultValue(instance, property, configAttr);
        }
    }

    private void SetDefaultValue(object instance, PropertyInfo property, ConfigurationAttribute configAttr)
    {
        if (configAttr.DefaultValue != null)
        {
            try
            {
                var convertedDefault = ConvertValue(configAttr.DefaultValue, property.PropertyType);
                property.SetValue(instance, convertedDefault);
                Console.WriteLine($"デフォルト値設定: {configAttr.Key} = {convertedDefault}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"デフォルト値設定エラー {configAttr.Key}: {ex.Message}");
            }
        }
    }

    private object ConvertValue(object value, Type targetType)
    {
        if (value == null) return null;
        if (targetType.IsAssignableFrom(value.GetType())) return value;

        // Nullable型の処理
        if (targetType.IsGenericType && targetType.GetGenericTypeDefinition() == typeof(Nullable<>))
        {
            var underlyingType = Nullable.GetUnderlyingType(targetType);
            return ConvertValue(value, underlyingType);
        }

        // Enum型の処理
        if (targetType.IsEnum)
        {
            return Enum.Parse(targetType, value.ToString(), true);
        }

        // 基本的な型変換
        return Convert.ChangeType(value, targetType);
    }

public class SimpleORM
{
    private readonly string _connectionString;

    public SimpleORM(string connectionString)
    {
        _connectionString = connectionString;
    }

    public string GenerateCreateTableSQL<T>()
    {
        var type = typeof(T);
        var tableAttr = type.GetCustomAttribute<TableAttribute>();

        if (tableAttr == null)
            throw new InvalidOperationException($"{type.Name}にTable属性が定義されていません");

        var tableName = $"{tableAttr.Schema}.{tableAttr.Name}";
        var columns = new List<string>();

        foreach (var property in type.GetProperties())
        {
            var columnAttr = property.GetCustomAttribute<ColumnAttribute>();
            if (columnAttr != null)
            {
                var columnDefinition = GenerateColumnDefinition(property, columnAttr);
                columns.Add(columnDefinition);
            }
        }

        var sql = $"CREATE TABLE {tableName} (\n  {string.Join(",\n  ", columns)}\n);";
        return sql;
    }

    private string GenerateColumnDefinition(PropertyInfo property, ColumnAttribute columnAttr)
    {
        var columnName = columnAttr.Name ?? property.Name;
        var dataType = columnAttr.DataType ?? GetSqlDataType(property.PropertyType);

        var definition = $"{columnName} {dataType}";

        if (columnAttr.IsIdentity)
        {
            definition += " IDENTITY(1,1)";
        }

        if (columnAttr.IsPrimaryKey)
        {
            definition += " PRIMARY KEY";
        }

        if (columnAttr.IsRequired)
        {
            definition += " NOT NULL";
        }

        return definition;
    }

    private string GetSqlDataType(Type propertyType)
    {
        // Nullable型の処理
        if (propertyType.IsGenericType && propertyType.GetGenericTypeDefinition() == typeof(Nullable<>))
        {
            propertyType = Nullable.GetUnderlyingType(propertyType);
        }

        return propertyType.Name switch
        {
            nameof(Int32) => "INT",
            nameof(Int64) => "BIGINT",
            nameof(String) => "NVARCHAR(255)",
            nameof(DateTime) => "DATETIME2",
            nameof(Decimal) => "DECIMAL(18,2)",
            nameof(Boolean) => "BIT",
            nameof(Guid) => "UNIQUEIDENTIFIER",
            _ => "NVARCHAR(255)"
        };
    }

    public string GenerateSelectSQL<T>(object id = null)
    {
        var type = typeof(T);
        var tableAttr = type.GetCustomAttribute<TableAttribute>();
        var tableName = $"{tableAttr.Schema}.{tableAttr.Name}";

        var columns = GetColumnNames(type);
        var sql = $"SELECT {string.Join(", ", columns)} FROM {tableName}";

        if (id != null)
        {
            var primaryKeyColumn = GetPrimaryKeyColumn(type);
            sql += $" WHERE {primaryKeyColumn} = @Id";
        }

        return sql;
    }

    public string GenerateInsertSQL<T>()
    {
        var type = typeof(T);
        var tableAttr = type.GetCustomAttribute<TableAttribute>();
        var tableName = $"{tableAttr.Schema}.{tableAttr.Name}";

        var insertColumns = new List<string>();
        var valueParameters = new List<string>();

        foreach (var property in type.GetProperties())
        {
            var columnAttr = property.GetCustomAttribute<ColumnAttribute>();
            if (columnAttr != null && !columnAttr.IsIdentity)
            {
                insertColumns.Add(columnAttr.Name ?? property.Name);
                valueParameters.Add($"@{property.Name}");
            }
        }

        return $"INSERT INTO {tableName} ({string.Join(", ", insertColumns)}) " +
               $"VALUES ({string.Join(", ", valueParameters)})";
    }

    private List<string> GetColumnNames(Type type)
    {
        return type.GetProperties()
            .Where(p => p.GetCustomAttribute<ColumnAttribute>() != null)
            .Select(p => p.GetCustomAttribute<ColumnAttribute>().Name ?? p.Name)
            .ToList();
    }

    private string GetPrimaryKeyColumn(Type type)
    {
        var primaryKeyProperty = type.GetProperties()
            .FirstOrDefault(p => p.GetCustomAttribute<ColumnAttribute>()?.IsPrimaryKey == true);

        if (primaryKeyProperty == null)
            throw new InvalidOperationException($"{type.Name}にプライマリキーが定義されていません");

        var columnAttr = primaryKeyProperty.GetCustomAttribute<ColumnAttribute>();
        return columnAttr.Name ?? primaryKeyProperty.Name;
    }

    public void GenerateEntityDocumentation<T>()

                Console.WriteLine($"    データ型: {columnAttr.DataType ?? GetSqlDataType(property.PropertyType)}");

    public class ReflectionPerformanceOptimizer
{
    // キャッシュによる最適化
    private static readonly ConcurrentDictionary<Type, PropertyInfo[]> _propertyCache = new();
    private static readonly ConcurrentDictionary<Type, MethodInfo[]> _methodCache = new();
    private static readonly ConcurrentDictionary<string, Delegate> _compiledAccessors = new();

    public void DemonstrateOptimizations()
    {
        Console.WriteLine("=== リフレクション最適化技法 ===");

        // 1. キャッシュの効果
        DemonstrateCaching();

        // 2. 式木コンパイルの効果
        DemonstrateExpressionCompilation();

        // 3. 最適化手法の比較
        ComparePerformance();
    }

    private void DemonstrateCaching()
    {
        Console.WriteLine("\n--- キャッシュ最適化 ---");

        const int iterations = 10000;
        var employee = new Employee("テスト", "部署");

        // キャッシュなし
        var sw = Stopwatch.StartNew();
        for (int i = 0; i < iterations; i++)
        {
            var properties = typeof(Employee).GetProperties(); // 毎回取得
        }
        sw.Stop();
        Console.WriteLine($"キャッシュなし: {sw.ElapsedMilliseconds}ms");

        // キャッシュあり
        sw.Restart();
        for (int i = 0; i < iterations; i++)
        {
            var properties = GetCachedProperties(typeof(Employee));
        }
        sw.Stop();
        Console.WriteLine($"キャッシュあり: {sw.ElapsedMilliseconds}ms");
    }

    private PropertyInfo[] GetCachedProperties(Type type)
    {
        return _propertyCache.GetOrAdd(type, t => t.GetProperties());
    }

    private void DemonstrateExpressionCompilation()
    {
        Console.WriteLine("\n--- 式木コンパイル最適化 ---");

        const int iterations = 100000;
        var employee = new Employee("テスト", "部署");
        var nameProperty = typeof(Employee).GetProperty("Name");

        // 通常のリフレクション
        var sw = Stopwatch.StartNew();
        for (int i = 0; i < iterations; i++)
        {
            var value = nameProperty.GetValue(employee);
        }
        sw.Stop();
        Console.WriteLine($"リフレクション: {sw.ElapsedMilliseconds}ms");

        // コンパイル済みアクセサー
        var getter = CreatePropertyGetter<Employee, string>("Name");
        sw.Restart();
        for (int i = 0; i < iterations; i++)
        {
            var value = getter(employee);
        }
        sw.Stop();
        Console.WriteLine($"コンパイル済み: {sw.ElapsedMilliseconds}ms");

        // 直接アクセス（参考）
        sw.Restart();
        for (int i = 0; i < iterations; i++)
        {
            var value = employee.Name;
        }
        sw.Stop();
        Console.WriteLine($"直接アクセス: {sw.ElapsedMilliseconds}ms");
    }

    private Func<T, TProperty> CreatePropertyGetter<T, TProperty>(string propertyName)
    {
        var cacheKey = $"{typeof(T).FullName}.{propertyName}.Get";

        return (Func<T, TProperty>)_compiledAccessors.GetOrAdd(cacheKey, _ =>
        {
            var parameter = Expression.Parameter(typeof(T), "obj");
            var property = Expression.Property(parameter, propertyName);
            var lambda = Expression.Lambda<Func<T, TProperty>>(property, parameter);
            return lambda.Compile();
        });
    }

    private void ComparePerformance()
    {
        Console.WriteLine("\n--- 総合パフォーマンス比較 ---");

        var testCases = new[]
        {
            ("プロパティ取得", (Action)(() => TestPropertyGetting())),
            ("メソッド呼び出し", (Action)(() => TestMethodInvocation())),
            ("オブジェクト生成", (Action)(() => TestObjectCreation()))
        };

        foreach (var (name, action) in testCases)

        }
    }

    public class ReflectionSecurityManager
{
    // 許可されたアセンブリのリスト
    private readonly HashSet<string> _allowedAssemblies;

    // 許可されたメソッドのパターン
    private readonly HashSet<string> _allowedMethodPatterns;

    // 禁止されたタイプ
    private readonly HashSet<Type> _forbiddenTypes;

    public ReflectionSecurityManager()
    {
        _allowedAssemblies = new HashSet<string>
        {
            "System.Private.CoreLib",
            "MyApplication", // 自分のアプリケーション
            "MyApplication.Domain"
        };

        _allowedMethodPatterns = new HashSet<string>
        {
            "Get*",
            "Set*",
            "ToString",
            "Equals",
            "GetHashCode"
        };

        _forbiddenTypes = new HashSet<Type>
        {
            typeof(System.IO.File),
            typeof(System.Diagnostics.Process),
            typeof(System.Reflection.Assembly)
        };
    }

    public void DemonstrateSecurityChecks()
    {
        Console.WriteLine("=== リフレクションセキュリティチェック ===");

        // 安全なタイプアクセス
        TestSafeTypeAccess();

        // 危険なメソッド呼び出しの防止
        TestDangerousMethodPrevention();

        // アセンブリ制限
        TestAssemblyRestriction();
    }

    private void TestSafeTypeAccess()
    {
        Console.WriteLine("\n--- 安全なタイプアクセス ---");

        var safeType = typeof(Employee);
        var dangerousType = typeof(System.IO.File);

        Console.WriteLine($"Employee型のアクセス: {IsTypeAccessAllowed(safeType)}");
        Console.WriteLine($"File型のアクセス: {IsTypeAccessAllowed(dangerousType)}");
    }

    private void TestDangerousMethodPrevention()
    {
        Console.WriteLine("\n--- 危険なメソッド呼び出しの防止 ---");

        try
        {
            // 安全なメソッド
            InvokeMethodSafely(typeof(Employee), "GetEmployeeInfo", new Employee("テスト", "部署"), null);

            // 危険なメソッド（実際には呼び出されない）
            InvokeMethodSafely(typeof(System.IO.File), "Delete", null, new object[] { "somefile.txt" });
        }
        catch (SecurityException ex)
        {
            Console.WriteLine($"セキュリティ例外: {ex.Message}");
        }
    }

    private void TestAssemblyRestriction()
    {
        Console.WriteLine("\n--- アセンブリ制限 ---");

        var currentAssembly = typeof(Employee).Assembly;
        var systemAssembly = typeof(string).Assembly;

        Console.WriteLine($"アプリケーションアセンブリ: {IsAssemblyAllowed(currentAssembly)}");
        Console.WriteLine($"システムアセンブリ: {IsAssemblyAllowed(systemAssembly)}");
    }

    public bool IsTypeAccessAllowed(Type type)
    {
        // 禁止タイプのチェック
        if (_forbiddenTypes.Contains(type))
        {
            return false;
        }

        // アセンブリのチェック
        if (!IsAssemblyAllowed(type.Assembly))
        {
            return false;
        }

        // 危険な名前空間のチェック
        var dangerousNamespaces = new[]
        {
            "System.IO",
            "System.Diagnostics",
            "System.Net",
            "System.Security"
        };

        if (dangerousNamespaces.Any(ns => type.Namespace?.StartsWith(ns) == true))
        {
            return false;
        }

        return true;
    }

    public bool IsAssemblyAllowed(Assembly assembly)
    {
        var assemblyName = assembly.GetName().Name;

        // 許可リストにあるか確認
        if (_allowedAssemblies.Contains(assemblyName))
        {
            return true;
        }

        // アプリケーション固有のアセンブリは許可
        if (assemblyName?.StartsWith("MyApplication") == true)
        {
            return true;
        }

        return false;
    }

    public object InvokeMethodSafely(Type type, string methodName, object instance, object[] parameters)
    {
        // タイプアクセスチェック
        if (!IsTypeAccessAllowed(type))
        {
            throw new SecurityException($"タイプ {type.Name} へのアクセスは許可されていません");
        }

        // メソッド名パターンチェック
        if (!IsMethodNameAllowed(methodName))
        {
            throw new SecurityException($"メソッド {methodName} の実行は許可されていません");
        }

        var method = type.GetMethod(methodName);
        if (method == null)
        {
            throw new MethodAccessException($"メソッド {methodName} が見つかりません");
        }

        // 追加のセキュリティチェック
        if (IsMethodDangerous(method))
        {
            throw new SecurityException($"危険なメソッド {methodName} の実行は禁止されています");
        }

        Console.WriteLine($"安全なメソッド実行: {type.Name}.{methodName}");
        return method.Invoke(instance, parameters);
    }

    private bool IsMethodNameAllowed(string methodName)
    {
        return _allowedMethodPatterns.Any(pattern =>
        {
            if (pattern.EndsWith("*"))
            {
                return methodName.StartsWith(pattern.TrimEnd('*'));
            }
            return methodName == pattern;
        });
    }

    private bool IsMethodDangerous(MethodInfo method)
    {
        var dangerousMethodNames = new[]
        {
            "Delete",
            "Execute",
            "Start",
            "Kill",
            "Load",
            "LoadFrom"
        };

        return dangerousMethodNames.Contains(method.Name);
    }

    // 入力検証とサニタイズ
    public string SanitizeInput(string input)
    {
        if (string.IsNullOrEmpty(input))
            return input;

        // 危険な文字の除去
        var dangerousChars = new[] { ';', '\'', '"', '<', '>', '&' };
        foreach (var ch in dangerousChars)
        {
            input = input.Replace(ch.ToString(), "");
        }

        return input;
    }

    // プロパティアクセスの制限
    public object GetPropertySafely(Type type, string propertyName, object instance)
    {
        if (!IsTypeAccessAllowed(type))
        {
            throw new SecurityException($"タイプ {type.Name} へのアクセスは許可されていません");
        }

        propertyName = SanitizeInput(propertyName);

        var property = type.GetProperty(propertyName);
        if (property == null)
        {
            throw new ArgumentException($"プロパティ {propertyName} が見つかりません");
        }

        if (!property.CanRead)
        {
            throw new InvalidOperationException($"プロパティ {propertyName} は読み取り専用ではありません");
        }

        return property.GetValue(instance);
    }
