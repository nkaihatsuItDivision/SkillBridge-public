    using System;
using System.IO;

using System.Threading;

// 正しいIDisposableパターンの実装例
public class ResourceManager : IDisposable
{
    private readonly FileStream _fileStream;
    private readonly HttpClient _httpClient;
    private readonly Timer _timer;
    private bool _disposed = false;

    public ResourceManager(string filePath)
    {
        try
        {
            _fileStream = new FileStream(filePath, FileMode.OpenOrCreate, FileAccess.ReadWrite);
            _httpClient = new HttpClient { Timeout = TimeSpan.FromSeconds(30) };
            _timer = new Timer(OnTimerElapsed, null, TimeSpan.Zero, TimeSpan.FromSeconds(5));

            Console.WriteLine($"ResourceManager initialized for: {filePath}");
        }
        catch
        {
            // 初期化中にエラーが発生した場合、既に作成されたリソースを確実に解放
            _fileStream?.Dispose();
            _httpClient?.Dispose();
            _timer?.Dispose();
            throw;
        }
    }

    private void OnTimerElapsed(object state)
    {
        if (!_disposed)
        {
            Console.WriteLine($"Timer tick: {DateTime.Now:HH:mm:ss} - Memory: {GC.GetTotalMemory(false):N0} bytes");
        }
    }

    public void WriteData(string data)
    {
        ThrowIfDisposed();

        var bytes = System.Text.Encoding.UTF8.GetBytes(data + Environment.NewLine);
        _fileStream.Write(bytes, 0, bytes.Length);
        _fileStream.Flush();

        Console.WriteLine($"Data written: {data}");
    }

    public async Task<string> FetchDataAsync(string url)
    {
        ThrowIfDisposed();

        try
        {
            var response = await _httpClient.GetStringAsync(url);
            Console.WriteLine($"Data fetched from: {url}");
            return response;
        }
        catch (HttpRequestException ex)
        {
            Console.WriteLine($"HTTP error: {ex.Message}");
            throw;
        }
    }

    // パブリックなDisposeメソッド
    public void Dispose()
    {
        Dispose(disposing: true);
        GC.SuppressFinalize(this); // ファイナライザーの実行を抑制
    }

    // 実際のリソース解放処理（継承を考慮した設計）
    protected virtual void Dispose(bool disposing)
    {
        if (!_disposed)
        {
            if (disposing)
            {
                // マネージドリソースの解放
                Console.WriteLine("Disposing managed resources...");
                _fileStream?.Dispose();
                _httpClient?.Dispose();
                _timer?.Dispose();
            }

            // アンマネージドリソースの解放（この例では該当なし）
            Console.WriteLine("Disposing unmanaged resources...");

            _disposed = true;
        }
    }

    // ファイナライザー（通常は実装不要だが、アンマネージドリソースがある場合のパターン）
    ~ResourceManager()
    {
        Console.WriteLine("Finalizer called - this should be rare!");
        Dispose(disposing: false);
    }

    private void ThrowIfDisposed()
    {
        if (_disposed)
        {
            throw new ObjectDisposedException(nameof(ResourceManager));
        }
    }
}

    {
        _id = id;
    }

    {

            throw new ObjectDisposedException(nameof(SafeEventSubscriber));

    }

    private void OnDataChanged(string data)

        Console.WriteLine($"Safe subscriber {_id} received: {data}");

    {

        {

            {

                _publisher = null;
            }

            _disposed = true;
        }
    }
}

public class WeakEventManager
{
    private readonly List<WeakReference> _subscribers = new();
    private readonly object _lock = new object();

    public void Subscribe(Action<string> handler)
    {
        lock (_lock)
        {
            _subscribers.Add(new WeakReference(handler));
        }
    }

    public void Publish(string data)
    {
        List<WeakReference> aliveSubscribers;

        lock (_lock)
        {
            aliveSubscribers = new List<WeakReference>();

            foreach (var weakRef in _subscribers)
            {
                if (weakRef.Target is Action<string> handler)
                {
                    try
                    {
                        handler(data);
                        aliveSubscribers.Add(weakRef);
                    }
                    catch (Exception ex)
                    {
                        Console.WriteLine($"Handler execution failed: {ex.Message}");
                    }
                }
                // 既にGCされたハンドラーは自動的に除外される
            }

            _subscribers.Clear();
            _subscribers.AddRange(aliveSubscribers);
        }

        Console.WriteLine($"Event published to {aliveSubscribers.Count} active subscribers");
    }

    public int GetActiveSubscriberCount()
    {
        lock (_lock)
        {
            return _subscribers.Count(wr => wr.IsAlive);
        }
    }
}

    using System;
using System.Buffers;
using System.Collections.Concurrent;

public class LargeObjectOptimizer
{
    private const int LOH_THRESHOLD = 85000; // 85KB以上はLOHに配置
    private readonly ConcurrentQueue<byte[]> _arrayPool = new();
    private int _poolHits = 0;
    private int _poolMisses = 0;

    // 問題のあるコード：大きな配列を頻繁に作成・破棄
    public void DemonstrateLOHProblem()
    {
        Console.WriteLine("=== LOH問題のデモンストレーション ===");

        var beforeMemory = GC.GetTotalMemory(true);
        var stopwatch = System.Diagnostics.Stopwatch.StartNew();

        for (int i = 0; i < 1000; i++)
        {
            // 100KBの配列を作成（LOHに配置される）
            var largeArray = new byte[100 * 1024];

            // 簡単な処理をシミュレート
            ProcessLargeArray(largeArray);

            if (i % 100 == 0)
            {
                var currentMemory = GC.GetTotalMemory(false);
                Console.WriteLine($"Iteration {i}: Memory usage = {currentMemory - beforeMemory:N0} bytes");
            }
        }

        stopwatch.Stop();
        var afterMemory = GC.GetTotalMemory(false);

        Console.WriteLine($"Problem approach completed in {stopwatch.ElapsedMilliseconds}ms");
        Console.WriteLine($"Memory delta: {afterMemory - beforeMemory:N0} bytes");
        Console.WriteLine("Large arrays created. Memory may not be immediately freed due to LOH behavior.");
    }

    // 改善版：カスタムArrayPoolの使用
    public void DemonstrateArrayPooling()
    {
        Console.WriteLine("\n=== Array Pooling最適化デモ ===");

        var beforeMemory = GC.GetTotalMemory(true);
        var stopwatch = System.Diagnostics.Stopwatch.StartNew();

        for (int i = 0; i < 1000; i++)
        {
            var array = RentArray(100 * 1024);

            ProcessLargeArray(array);

            ReturnArray(array);

            if (i % 100 == 0)
            {
                var currentMemory = GC.GetTotalMemory(false);
                Console.WriteLine($"Iteration {i}: Memory = {currentMemory - beforeMemory:N0} bytes, " +
                                 $"Pool hits: {_poolHits}, misses: {_poolMisses}");
            }
        }

        stopwatch.Stop();
        var afterMemory = GC.GetTotalMemory(false);

        Console.WriteLine($"Pooling approach completed in {stopwatch.ElapsedMilliseconds}ms");
        Console.WriteLine($"Memory delta: {afterMemory - beforeMemory:N0} bytes");
        Console.WriteLine($"Final pool size: {_arrayPool.Count} arrays");
        Console.WriteLine($"Pool efficiency: {_poolHits}/{_poolHits + _poolMisses} hits");
    }

    public byte[] RentArray(int minimumSize)
    {
        if (minimumSize >= LOH_THRESHOLD)
        {
            // プールから再利用可能な配列を探す
            if (_arrayPool.TryDequeue(out var pooledArray) && pooledArray.Length >= minimumSize)
            {
                _poolHits++;
                return pooledArray;
            }
        }

        _poolMisses++;
        return new byte[minimumSize];
    }

    public void ReturnArray(byte[] array)
    {
        if (array.Length >= LOH_THRESHOLD)
        {
            // 配列をクリアして（セキュリティのため）プールに戻す
            Array.Clear(array, 0, array.Length);
            _arrayPool.Enqueue(array);
        }
    }

    // .NET標準のArrayPool<T>を使用した最適化例
    public void UsingSystemArrayPool()
    {
        Console.WriteLine("\n=== System.ArrayPool使用例 ===");

        var pool = ArrayPool<byte>.Shared;
        var beforeMemory = GC.GetTotalMemory(true);
        var stopwatch = System.Diagnostics.Stopwatch.StartNew();

        for (int i = 0; i < 1000; i++)
        {
            // 配列を借りる（実際のサイズは要求より大きい可能性がある）
            var array = pool.Rent(100 * 1024);

            try
            {
                ProcessLargeArray(array);

                if (i % 100 == 0)
                {
                    Console.WriteLine($"ArrayPool iteration {i}: Array size = {array.Length:N0}");
                }
            }
            finally
            {
                // 必ず返却する（clearArrayでデータをクリア）
                pool.Return(array, clearArray: true);
            }
        }

        stopwatch.Stop();
        var afterMemory = GC.GetTotalMemory(false);

        Console.WriteLine($"ArrayPool approach completed in {stopwatch.ElapsedMilliseconds}ms");
        Console.WriteLine($"Memory delta: {afterMemory - beforeMemory:N0} bytes");
    }

    private void ProcessLargeArray(byte[] array)
    {
        // 配列処理をシミュレート
        int checksum = 0;
        for (int i = 0; i < Math.Min(1000, array.Length); i++)
        {
            array[i] = (byte)(i % 256);
            checksum += array[i];
        }
    }

    using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Text;

public static class PerformanceMeasurement
{
    public static TimeSpan MeasureExecutionTime(Action action, int iterations = 1)
    {
        // ウォームアップ実行（JITコンパイルの影響を除去）
        action();

        // GCの影響を最小化
        GC.Collect();
        GC.WaitForPendingFinalizers();
        GC.Collect();

        var stopwatch = Stopwatch.StartNew();

        for (int i = 0; i < iterations; i++)
        {
            action();
        }

        stopwatch.Stop();
        return stopwatch.Elapsed;
    }

    public static (TimeSpan time, long memoryUsed, int gen0, int gen1, int gen2) MeasureTimeAndMemory(
        Action action, int iterations = 1)
    {
        var gcBefore = (GC.CollectionCount(0), GC.CollectionCount(1), GC.CollectionCount(2));
        var memoryBefore = GC.GetTotalMemory(true);

        var time = MeasureExecutionTime(action, iterations);

        var memoryAfter = GC.GetTotalMemory(false);
        var gcAfter = (GC.CollectionCount(0), GC.CollectionCount(1), GC.CollectionCount(2));

        return (
            time, 
            memoryAfter - memoryBefore,
            gcAfter.Item1 - gcBefore.Item1,
            gcAfter.Item2 - gcBefore.Item2,
            gcAfter.Item3 - gcBefore.Item3
        );
    }

    public static void ComparePerformance(
        string method1Name, Action method1,
        string method2Name, Action method2,
        int iterations = 1000)
    {
        Console.WriteLine($"=== パフォーマンス比較: {iterations:N0} 回実行 ===");

        var (time1, memory1, gen0_1, gen1_1, gen2_1) = MeasureTimeAndMemory(method1, iterations);
        var (time2, memory2, gen0_2, gen1_2, gen2_2) = MeasureTimeAndMemory(method2, iterations);

        Console.WriteLine($"\n📊 {method1Name}:");
        Console.WriteLine($"  実行時間: {time1.TotalMilliseconds:F2} ms");
        Console.WriteLine($"  メモリ使用量: {memory1:N0} bytes");
        Console.WriteLine($"  GC実行回数: Gen0={gen0_1}, Gen1={gen1_1}, Gen2={gen2_1}");

        Console.WriteLine($"\n📊 {method2Name}:");
        Console.WriteLine($"  実行時間: {time2.TotalMilliseconds:F2} ms");
        Console.WriteLine($"  メモリ使用量: {memory2:N0} bytes");
        Console.WriteLine($"  GC実行回数: Gen0={gen0_2}, Gen1={gen1_2}, Gen2={gen2_2}");

        // 改善度の計算と表示
        var speedImprovement = time1.TotalMilliseconds / time2.TotalMilliseconds;
        var memoryImprovement = memory1 != 0 ? (double)memory1 / memory2 : double.NaN;

        Console.WriteLine($"\n🚀 パフォーマンス改善:");
        Console.WriteLine($"  速度: {method2Name}は{method1Name}の{speedImprovement:F2}倍の速度");

        if (!double.IsNaN(memoryImprovement) && memoryImprovement > 0)
        {
            Console.WriteLine($"  メモリ効率: {method2Name}は{method1Name}の{memoryImprovement:F2}倍効率的");
        }
    }

    // 統計的に有意な測定を行う
    public static PerformanceStatistics MeasureWithStatistics(Action action, int runs = 10)
    {
        var times = new List<double>();
        var memories = new List<long>();

        for (int i = 0; i < runs; i++)
        {
            var (time, memory, _, _, _) = MeasureTimeAndMemory(action);
            times.Add(time.TotalMilliseconds);
            memories.Add(memory);
        }

        return new PerformanceStatistics
        {
            TimeAverage = times.Average(),
            TimeStdDev = CalculateStandardDeviation(times),
            TimeMin = times.Min(),
            TimeMax = times.Max(),
            MemoryAverage = memories.Average(),
            MemoryStdDev = CalculateStandardDeviation(memories.Select(m => (double)m)),
            Runs = runs
        };
    }

    private static double CalculateStandardDeviation(IEnumerable<double> values)
    {
        var enumerable = values.ToList();
        var average = enumerable.Average();
        var sumOfSquaresOfDifferences = enumerable.Select(val => (val - average) * (val - average)).Sum();
        return Math.Sqrt(sumOfSquaresOfDifferences / enumerable.Count);
    }
}

    public async Task<int> ProcessLargeDataSetAsync<T>(
        IAsyncEnumerable<T> dataSource,
        Func<T, Task<string>> processor,
        int maxConcurrency = Environment.ProcessorCount)
    {
        var processed = 0;
        var semaphore = new SemaphoreSlim(maxConcurrency, maxConcurrency);
        var processingTasks = new List<Task>();

        await foreach (var item in dataSource)
        {
            await semaphore.WaitAsync();

            var task = ProcessItemAsync(item, processor, semaphore);
            processingTasks.Add(task);

            // 完了したタスクを定期的に除去（メモリ効率）
            if (processingTasks.Count >= maxConcurrency * 2)
            {

                foreach (var completedTask in completedTasks)
                {
                    processingTasks.Remove(completedTask);
                    processed++;
                }
            }
        }

        // 残りのタスクの完了を待機
        await Task.WhenAll(processingTasks);
        processed += processingTasks.Count;

        Console.WriteLine($"Total items processed: {processed}");
        return processed;
    }

    2. GC制御: 測定前にフルGCを実行
    3. 統計的有意性: 複数回実行して平均と標準偏差を算出
    4. 環境の一貫性: 同一環境・同一条件での測定
    5. 実用的な負荷: 本番に近いデータサイズと処理パターン

    最適化の判断基準

    いつ最適化すべきか

    • パフォーマンス要件に達していない場合
    • メモリ使用量が予算を超過している場合
    • スケーラビリティに問題がある場合
    • ユーザー体験に悪影響がある場合

    いつ最適化すべきでないか

    • 測定していない推測に基づく最適化
    • 読みやすさを大幅に損なう場合
    • 保守性とのトレードオフが悪い場合
    • ビジネス価値が不明確な場合

    メモリ管理とパフォーマンス最適化の価値

    現代のソフトウェア開発において、メモリ効率とパフォーマンスは技術的な優秀さを示すだけでなく、ビジネス競争力の源泉となります。

    学習した核心技術の統合活用

    1. ガベージコレクション最適化: 安定したメモリ使用パターンの確立
    2. メモリリーク防止: 長期稼働における信頼性の確保
    3. 科学的測定: 推測ではなく数値に基づく改善
    4. 最新技術活用: Span、Memoryによる次世代最適化
    5. 非同期最適化: スケーラブルで応答性の高いアプリケーション

    継続的改善の文化

    • 測定ファースト: すべての最適化は測定から始まる
    • 段階的改善: 小さな改善の積み重ねが大きな成果を生む
    • チーム共有: 最適化技法は個人の知識ではなくチームの資産
    • ビジネス価値重視: 技術的な美しさより実用的な価値を優先
