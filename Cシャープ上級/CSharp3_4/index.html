    はじめに - 高度な例外処理の重要性

    例外処理は、堅牢なアプリケーションを構築するための最も重要な技術の一つです。基本的なtry-catchだけでは対応できない複雑なエラーシナリオに対処するため、高度な技法が必要となります。

    なぜ高度な例外処理が必要なのか

    • システムの信頼性向上: 予期しないエラーからの適切な回復
    • 運用効率の改善: 詳細なログとモニタリングによる迅速な問題解決
    • ユーザー体験の向上: 分かりやすいエラーメッセージと適切な誘導
    • 開発生産性: 統一されたエラーハンドリングパターンによる効率的な開発

    {

            ? binder(Value)

    }
}

    3. Retryパターン - 一時的障害からの自動回復

    技法の目的

    ネットワークエラーやリソース不足など、一時的な障害からの自動回復を実現します。

    実際の使用例

    • 外部API呼び出しの再試行
    • データベース接続の復旧処理
    • ファイルアクセスの競合回避
    • クラウドサービスの一時的障害対応

    汎用的なRetryPolicyの実装

    public class RetryPolicy
{
    private readonly int _maxRetries;
    private readonly TimeSpan _baseDelay;
    private readonly Func<Exception, bool> _shouldRetry;
    private readonly Func<int, TimeSpan> _delayStrategy;

    public RetryPolicy(
        int maxRetries = 3, 
        TimeSpan? baseDelay = null,
        Func<Exception, bool> shouldRetry = null,
        RetryStrategy strategy = RetryStrategy.FixedDelay)
    {
        _maxRetries = maxRetries;
        _baseDelay = baseDelay ?? TimeSpan.FromSeconds(1);
        _shouldRetry = shouldRetry ?? DefaultShouldRetry;
        _delayStrategy = GetDelayStrategy(strategy);
    }

    // デフォルトの再試行判定
    private static bool DefaultShouldRetry(Exception ex)
    {
        return ex is HttpRequestException ||
               ex is TaskCanceledException ||
               ex is SocketException ||
               ex is IOException;
    }

    // 遅延戦略の取得
    private Func<int, TimeSpan> GetDelayStrategy(RetryStrategy strategy)
    {
        return strategy switch
        {
            RetryStrategy.FixedDelay => attempt => _baseDelay,
            RetryStrategy.LinearBackoff => attempt => TimeSpan.FromMilliseconds(_baseDelay.TotalMilliseconds * (attempt + 1)),
            RetryStrategy.ExponentialBackoff => attempt => TimeSpan.FromMilliseconds(_baseDelay.TotalMilliseconds * Math.Pow(2, attempt)),
            RetryStrategy.RandomJitter => attempt => TimeSpan.FromMilliseconds(_baseDelay.TotalMilliseconds + Random.Shared.Next(0, 1000)),
            _ => attempt => _baseDelay
        };
    }

    // 同期版の実行
    public T Execute<T>(Func<T> operation, string operationName = null)
    {
        Exception lastException = null;

        for (int attempt = 0; attempt <= _maxRetries; attempt++)
        {
            try
            {
                return operation();
            }
            catch (Exception ex) when (_shouldRetry(ex) && attempt < _maxRetries)
            {
                lastException = ex;

                var delay = _delayStrategy(attempt);
                Console.WriteLine($"{operationName ?? "操作"} 失敗 (試行 {attempt + 1}/{_maxRetries + 1}): {ex.Message}");
                Console.WriteLine($"{delay.TotalSeconds}秒後に再試行します...");

                Thread.Sleep(delay);
            }
        }

        throw new AggregateException($"操作が {_maxRetries + 1} 回失敗しました: {operationName}", lastException);
    }

    // 非同期版の実行
    public async Task<T> ExecuteAsync<T>(Func<Task<T>> operation, string operationName = null)
    {
        Exception lastException = null;

        for (int attempt = 0; attempt <= _maxRetries; attempt++)
        {
            try
            {
                return await operation();
            }
            catch (Exception ex) when (_shouldRetry(ex) && attempt < _maxRetries)
            {
                lastException = ex;

                var delay = _delayStrategy(attempt);
                Console.WriteLine($"{operationName ?? "操作"} 失敗 (試行 {attempt + 1}/{_maxRetries + 1}): {ex.Message}");
                Console.WriteLine($"{delay.TotalSeconds}秒後に再試行します...");

                await Task.Delay(delay);
            }
        }

        throw new AggregateException($"操作が {_maxRetries + 1} 回失敗しました: {operationName}", lastException);
    }
}

public enum RetryStrategy
{
    FixedDelay,         // 固定間隔
    LinearBackoff,      // 線形増加
    ExponentialBackoff, // 指数関数的増加
    RandomJitter        // ランダムジッター
}

    実用的なRetryパターンの活用例

    public class ApiClient
{
    private readonly HttpClient _httpClient;
    private readonly RetryPolicy _retryPolicy;

    public ApiClient(HttpClient httpClient = null)
    {
        _httpClient = httpClient ?? new HttpClient();

        // API呼び出し用のRetryポリシー
        _retryPolicy = new RetryPolicy(
            maxRetries: 3,
            baseDelay: TimeSpan.FromSeconds(2),
            shouldRetry: ShouldRetryApiCall,
            strategy: RetryStrategy.ExponentialBackoff
        );
    }

    private bool ShouldRetryApiCall(Exception ex)
    {
        // HTTP例外の場合、ステータスコードによって判定
        if (ex is HttpRequestException httpEx)
        {
            // 一時的なエラーの場合のみ再試行
            return true; // 実際にはレスポンスコードを確認
        }

        // タイムアウトは再試行
        if (ex is TaskCanceledException timeoutEx && timeoutEx.InnerException is TimeoutException)
        {
            return true;
        }

        // ネットワークエラーは再試行
        return ex is SocketException || ex is IOException;
    }

    public async Task<Result<string>> GetDataAsync(string endpoint)
    {
        try
        {
            var result = await _retryPolicy.ExecuteAsync(async () =>
            {
                var response = await _httpClient.GetAsync(endpoint);
                response.EnsureSuccessStatusCode();
                return await response.Content.ReadAsStringAsync();
            }, $"API呼び出し: {endpoint}");

            return Result<string>.Success(result);
        }
        catch (AggregateException ex)
        {
            return Result<string>.Failure(ex.InnerException ?? ex, "API_CALL_FAILED");
        }
        catch (Exception ex)
        {
            return Result<string>.Failure(ex, "UNEXPECTED_ERROR");
        }
    }

    public async Task<Result<T>> PostDataAsync<T>(string endpoint, object data)
    {
        try
        {
            var result = await _retryPolicy.ExecuteAsync(async () =>
            {
                var json = JsonSerializer.Serialize(data);
                var content = new StringContent(json, Encoding.UTF8, "application/json");

                var response = await _httpClient.PostAsync(endpoint, content);
                response.EnsureSuccessStatusCode();

                var responseContent = await response.Content.ReadAsStringAsync();
                return JsonSerializer.Deserialize<T>(responseContent);
            }, $"API POST: {endpoint}");

            return Result<T>.Success(result);
        }
        catch (Exception ex)
        {
            return Result<T>.Failure(ex, "API_POST_FAILED");
        }
    }
}

    public class GlobalExceptionMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<GlobalExceptionMiddleware> _logger;
    private readonly IWebHostEnvironment _environment;

    public GlobalExceptionMiddleware(
        RequestDelegate next, 
        ILogger<GlobalExceptionMiddleware> logger,
        IWebHostEnvironment environment)
    {
        _next = next;
        _logger = logger;
        _environment = environment;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            await HandleExceptionAsync(context, ex);
        }
    }

    private async Task HandleExceptionAsync(HttpContext context, Exception exception)
    {
        // ログ記録
        var errorId = Guid.NewGuid().ToString("N")[..8];
        var logMessage = $"[{errorId}] {exception.GetType().Name}: {exception.Message}";

        _logger.LogError(exception, logMessage);

        // レスポンス設定
        context.Response.ContentType = "application/json";
        context.Response.StatusCode = GetStatusCode(exception);

        var response = CreateErrorResponse(exception, errorId);

        var jsonResponse = JsonSerializer.Serialize(response, new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            WriteIndented = _environment.IsDevelopment()
        });

        await context.Response.WriteAsync(jsonResponse);
    }

    private static int GetStatusCode(Exception exception) => exception switch
    {
        ValidationException => 400,
        AuthorizationException => 403,
        ResourceNotFoundException => 404,
        ExternalServiceException => 502,
        TimeoutException => 504,
        _ => 500
    };

    private object CreateErrorResponse(Exception exception, string errorId)
    {
        var baseResponse = new
        {
            error = new
            {
                id = errorId,
                message = GetUserFriendlyMessage(exception),
                timestamp = DateTime.UtcNow,
                type = exception.GetType().Name
            }
        };

        // 開発環境では詳細情報を追加
        if (_environment.IsDevelopment())
        {
            return new
            {
                error = new
                {
                    id = errorId,
                    message = GetUserFriendlyMessage(exception),
                    timestamp = DateTime.UtcNow,
                    type = exception.GetType().Name,
                    details = exception.Message,
                    stackTrace = exception.StackTrace,
                    data = GetExceptionData(exception)
                }
            };
        }

        return baseResponse;
    }

    private static string GetUserFriendlyMessage(Exception exception) => exception switch
    {
        ValidationException validationEx => "入力内容に問題があります。確認して再試行してください。",
        AuthorizationException => "この操作を実行する権限がありません。",
        ResourceNotFoundException => "要求されたリソースが見つかりません。",
        ExternalServiceException => "外部サービスとの通信でエラーが発生しました。しばらく時間をおいて再試行してください。",
        TimeoutException => "処理に時間がかかりすぎています。再試行してください。",
        _ => "システムエラーが発生しました。管理者にお問い合わせください。"
    };

    private static object GetExceptionData(Exception exception)
    {
        return exception switch
        {
            ValidationException validationEx => validationEx.ValidationErrors,
            BusinessException businessEx => new 
            { 
                businessEx.ErrorCode, 
                businessEx.Timestamp,
                businessEx.AdditionalData
            },
            _ => null
        };
    }
}

    public class GlobalExceptionHandler
{
    private readonly ILogger _logger;
    private readonly bool _isProductionEnvironment;

    public GlobalExceptionHandler(ILogger logger, bool isProductionEnvironment = false)
    {
        _logger = logger;
        _isProductionEnvironment = isProductionEnvironment;

        SetupExceptionHandlers();
    }

    private void SetupExceptionHandlers()
    {
        // アプリケーションドメインの未処理例外
        AppDomain.CurrentDomain.UnhandledException += OnUnhandledException;

        // UIスレッドの未処理例外
        Application.Current.DispatcherUnhandledException += OnDispatcherUnhandledException;

        // タスクの未処理例外
        TaskScheduler.UnobservedTaskException += OnUnobservedTaskException;
    }

    private void OnUnhandledException(object sender, UnhandledExceptionEventArgs e)
    {
        var exception = (Exception)e.ExceptionObject;
        var errorId = LogException(exception, "Critical unhandled exception");

        ShowCriticalErrorDialog(exception, errorId, e.IsTerminating);

        if (e.IsTerminating)
        {
            Environment.Exit(1);
        }
    }

    private void OnDispatcherUnhandledException(object sender, DispatcherUnhandledExceptionEventArgs e)
    {
        var errorId = LogException(e.Exception, "UI thread unhandled exception");

        if (ShouldHandleException(e.Exception))
        {
            ShowErrorDialog(e.Exception, errorId);
            e.Handled = true; // 例外を処理済みとしてアプリを継続
        }
        else
        {
            ShowCriticalErrorDialog(e.Exception, errorId, true);
        }
    }

    private void OnUnobservedTaskException(object sender, UnobservedTaskExceptionEventArgs e)
    {
        var errorId = LogException(e.Exception, "Unobserved task exception");
        e.SetObserved(); // 例外を観測済みとしてマーク

        // バックグラウンド処理のエラーはユーザーに通知（軽微な場合）
        if (ShouldNotifyUser(e.Exception))
        {
            ShowBackgroundErrorNotification(e.Exception, errorId);
        }
    }

    private string LogException(Exception exception, string context)
    {
        var errorId = Guid.NewGuid().ToString("N")[..8];
        var logMessage = $"[{errorId}] {context}: {exception.Message}";

        _logger.LogError(exception, logMessage);

        return errorId;
    }

    private bool ShouldHandleException(Exception exception)
    {
        // ビジネス例外は通常処理を継続可能
        return exception is BusinessException ||
               exception is ValidationException ||
               exception is HttpRequestException;
    }

    private bool ShouldNotifyUser(Exception exception)
    {
        // 重要度が高い例外のみユーザーに通知
        return exception is ExternalServiceException ||
               exception is IOException;
    }

    private void ShowErrorDialog(Exception exception, string errorId)
    {
        var message = GetUserFriendlyMessage(exception);
        var title = "エラーが発生しました";

        if (!_isProductionEnvironment)
        {
            message += $"\n\nエラーID: {errorId}\n詳細: {exception.Message}";
        }

        MessageBox.Show(message, title, MessageBoxButton.OK, MessageBoxImage.Warning);
    }

    private void ShowCriticalErrorDialog(Exception exception, string errorId, bool willTerminate)
    {
        var message = willTerminate 
            ? "重大なエラーが発生したため、アプリケーションを終了します。"
            : "重大なエラーが発生しました。";

        if (!_isProductionEnvironment)
        {
            message += $"\n\nエラーID: {errorId}\n詳細: {exception.Message}";
        }

        MessageBox.Show(message, "重大なエラー", MessageBoxButton.OK, MessageBoxImage.Error);
    }

    private void ShowBackgroundErrorNotification(Exception exception, string errorId)
    {
        // トースト通知やステータスバーでの軽微な通知
        Console.WriteLine($"バックグラウンド処理でエラーが発生しました (ID: {errorId})");
    }

    private string GetUserFriendlyMessage(Exception exception) => exception switch
    {
        ValidationException => "入力内容に問題があります。確認して再試行してください。",
        AuthorizationException => "この操作を実行する権限がありません。",
        ResourceNotFoundException => "要求されたデータが見つかりません。",
        ExternalServiceException => "外部サービスとの通信でエラーが発生しました。",
        TimeoutException => "処理に時間がかかりすぎています。再試行してください。",
        IOException => "ファイルの読み書きでエラーが発生しました。",
        UnauthorizedAccessException => "ファイルやリソースへのアクセス権限がありません。",
        _ => "予期しないエラーが発生しました。"
    };
}

    public interface IErrorHandlingService
{
    Task<Result<T>> ExecuteAsync<T>(Func<Task<T>> operation, string operationName = null, string context = null);
    Task<Result> ExecuteAsync(Func<Task> operation, string operationName = null, string context = null);
    Task LogErrorAsync(Exception exception, string context = null, Dictionary<string, object> additionalData = null);
    void HandleError(Exception exception, string userMessage = null, bool shouldNotifyUser = true);
    Task<ErrorStatistics> GetErrorStatisticsAsync(DateTime from, DateTime to);
}

public class ErrorHandlingService : IErrorHandlingService
{
    private readonly ILogger<ErrorHandlingService> _logger;
    private readonly RetryPolicy _defaultRetryPolicy;
    private readonly IErrorNotificationService _notificationService;
    private readonly IErrorMetricsCollector _metricsCollector;
    private readonly IErrorPersistenceService _persistenceService;

    public ErrorHandlingService(
        ILogger<ErrorHandlingService> logger,
        IErrorNotificationService notificationService = null,
        IErrorMetricsCollector metricsCollector = null,
        IErrorPersistenceService persistenceService = null)
    {
        _logger = logger;
        _notificationService = notificationService;
        _metricsCollector = metricsCollector;
        _persistenceService = persistenceService;

        _defaultRetryPolicy = new RetryPolicy(
            maxRetries: 2,
            baseDelay: TimeSpan.FromMilliseconds(500),
            strategy: RetryStrategy.ExponentialBackoff
        );
    }

    public async Task<Result<T>> ExecuteAsync<T>(Func<Task<T>> operation, string operationName = null, string context = null)
    {
        var stopwatch = Stopwatch.StartNew();
        var operationId = Guid.NewGuid().ToString("N")[..8];

        try
        {
            // メトリクス収集開始
            _metricsCollector?.RecordOperationStart(operationName, operationId);

            var result = await _defaultRetryPolicy.ExecuteAsync(operation, operationName);

            stopwatch.Stop();
            _metricsCollector?.RecordOperationSuccess(operationName, operationId, stopwatch.Elapsed);

            return Result<T>.Success(result);
        }
        catch (BusinessException businessEx)
        {
            stopwatch.Stop();
            await HandleBusinessExceptionAsync(businessEx, operationName, context, operationId, stopwatch.Elapsed);
            return Result<T>.Failure(businessEx.Message, businessEx.ErrorCode);
        }
        catch (Exception ex)
        {
            stopwatch.Stop();
            await HandleSystemExceptionAsync(ex, operationName, context, operationId, stopwatch.Elapsed);
            return Result<T>.Failure("システムエラーが発生しました。しばらく時間をおいて再試行してください。", "SYSTEM_ERROR");
        }
    }

    public async Task<Result> ExecuteAsync(Func<Task> operation, string operationName = null, string context = null)
    {
        var result = await ExecuteAsync(async () =>
        {
            await operation();
            return true;
        }, operationName, context);

        return result.IsSuccess ? Result.Success() : Result.Failure(result.ErrorMessage, result.ErrorCode);
    }

    public async Task LogErrorAsync(Exception exception, string context = null, Dictionary<string, object> additionalData = null)
    {
        var errorEntry = new ErrorEntry
        {
            Id = Guid.NewGuid(),
            Timestamp = DateTime.UtcNow,
            ExceptionType = exception.GetType().Name,
            Message = exception.Message,
            StackTrace = exception.StackTrace,
            Context = context,
            Severity = GetErrorSeverity(exception),
            AdditionalData = additionalData ?? new Dictionary<string, object>()
        };

        // 構造化ログ出力
        _logger.LogError(exception, 
            "Error logged - ID: {ErrorId}, Type: {ExceptionType}, Context: {Context}",
            errorEntry.Id, errorEntry.ExceptionType, context);

        // 永続化
        if (_persistenceService != null)
        {
            await _persistenceService.SaveErrorAsync(errorEntry);
        }

        // メトリクス収集
        _metricsCollector?.RecordError(errorEntry);

        // 重要度に応じた通知
        if (ShouldNotifyOperation(exception))
        {
            await _notificationService?.NotifyOperationTeamAsync(errorEntry);
        }
    }

    private async Task HandleBusinessExceptionAsync(BusinessException exception, string operationName, string context, string operationId, TimeSpan duration)
    {
        var additionalData = new Dictionary<string, object>
        {
            ["OperationName"] = operationName,
            ["OperationId"] = operationId,
            ["Duration"] = duration.TotalMilliseconds,
            ["ErrorCode"] = exception.ErrorCode
        };

        additionalData.AddRange(exception.AdditionalData);

        await LogErrorAsync(exception, context, additionalData);
        _metricsCollector?.RecordOperationFailure(operationName, operationId, duration, exception.ErrorCode);
    }

    private async Task HandleSystemExceptionAsync(Exception exception, string operationName, string context, string operationId, TimeSpan duration)
    {
        var additionalData = new Dictionary<string, object>
        {
            ["OperationName"] = operationName,
            ["OperationId"] = operationId,
            ["Duration"] = duration.TotalMilliseconds,
            ["IsCritical"] = IsCriticalError(exception)
        };

        await LogErrorAsync(exception, context, additionalData);
        _metricsCollector?.RecordOperationFailure(operationName, operationId, duration, "SYSTEM_ERROR");

        // 重大エラーの場合は即座にエスカレーション
        if (IsCriticalError(exception))
        {
            await _notificationService?.NotifyDevelopmentTeamAsync(new ErrorEntry
            {
                Id = Guid.NewGuid(),
                Timestamp = DateTime.UtcNow,
                ExceptionType = exception.GetType().Name,
                Message = exception.Message,
                Context = context,
                Severity = ErrorSeverity.Critical,
                AdditionalData = additionalData
            });
        }
    }

    public void HandleError(Exception exception, string userMessage = null, bool shouldNotifyUser = true)
    {
        var message = userMessage ?? GetUserFriendlyMessage(exception);
        var severity = GetErrorSeverity(exception);

        if (shouldNotifyUser)
        {
            NotifyUser(message, severity);
        }
    }

    public async Task<ErrorStatistics> GetErrorStatisticsAsync(DateTime from, DateTime to)
    {
        if (_persistenceService == null)
        {
            throw new InvalidOperationException("Error persistence service is not configured");
        }

        return await _persistenceService.GetErrorStatisticsAsync(from, to);
    }

    private ErrorSeverity GetErrorSeverity(Exception exception) => exception switch
    {
        ValidationException => ErrorSeverity.Warning,
        AuthorizationException => ErrorSeverity.Warning,
        ResourceNotFoundException => ErrorSeverity.Info,
        ExternalServiceException => ErrorSeverity.Error,
        BusinessException => ErrorSeverity.Warning,
        OutOfMemoryException => ErrorSeverity.Critical,
        StackOverflowException => ErrorSeverity.Critical,
        _ => ErrorSeverity.Error
    };

    private bool IsCriticalError(Exception exception) => exception switch
    {
        OutOfMemoryException => true,
        StackOverflowException => true,
        AccessViolationException => true,
        _ => false
    };

    private bool ShouldNotifyOperation(Exception exception)
    {
        return GetErrorSeverity(exception) >= ErrorSeverity.Error;
    }

    private string GetUserFriendlyMessage(Exception exception) => exception switch
    {
        ValidationException => "入力内容に問題があります。確認して再試行してください。",
        AuthorizationException => "この操作を実行する権限がありません。",
        ResourceNotFoundException => "要求されたデータが見つかりません。",
        ExternalServiceException => "外部サービスとの通信でエラーが発生しました。",
        TimeoutException => "処理に時間がかかりすぎています。再試行してください。",
        _ => "システムエラーが発生しました。管理者にお問い合わせください。"
    };

    private void NotifyUser(string message, ErrorSeverity severity)
    {
        // 実装例（実際のアプリケーションでは適切なUI表示メカニズムを使用）
        var severityText = severity switch
        {
            ErrorSeverity.Info => "情報",
            ErrorSeverity.Warning => "警告",
            ErrorSeverity.Error => "エラー",
            ErrorSeverity.Critical => "重大エラー",
            _ => "通知"
        };

        Console.WriteLine($"[{severityText}] {message}");
    }
}
