    学習目標

    • 主要なデザインパターンの目的と使用場面を理解する
    • C#での実装方法を段階的に習得する

    はじめに - デザインパターンとは

    デザインパターンは、ソフトウェア設計における「よくある問題に対する再利用可能な解決策」です。料理のレシピのように、特定の状況で使える「設計の型」を提供します。

    なぜデザインパターンが重要なのか

    • コードの再利用性: 似たような問題を効率的に解決
    • 保守性の向上: 標準的な構造で理解しやすい
    • 開発チームでの共通理解: パターン名で設計意図を伝達
    • 拡張性: 変更に強い柔軟な設計

    今回学習する5つのパターン： 1. Singleton - 唯一のインスタンスを保証 2. Factory - オブジェクト生成の複雑さを隠蔽 3. Observer - オブジェクト間の疎結合な通知 4. Strategy - アルゴリズムの動的な切り替え 5. Command - 操作のカプセル化

    1. Singletonパターン - 唯一のインスタンスを保証

    パターンの目的

    「システム全体で1つだけ存在すべきオブジェクト」を実現します。

    実際の使用例

    • データベース接続管理
    • ログ出力システム
    • アプリケーション設定
    • キャッシュシステム

    基本的な実装（スレッドセーフ版）

    using System;

// スレッドセーフなSingleton実装
public sealed class DatabaseConnection
{
    // Lazyを使用してスレッドセーフかつ遅延初期化を実現
    private static readonly Lazy<DatabaseConnection> _instance = 
        new Lazy<DatabaseConnection>(() => new DatabaseConnection());

    // privateコンストラクタで外部からのインスタンス化を防ぐ
    private DatabaseConnection()
    {
        ConnectionString = "Server=localhost;Database=MyApp;";
        Console.WriteLine("データベース接続を初期化しました");
    }

    // 唯一のアクセスポイント
    public static DatabaseConnection Instance => _instance.Value;

    public string ConnectionString { get; private set; }

    public void ExecuteQuery(string query)
    {
        Console.WriteLine($"クエリ実行: {query}");
    }
}

    実用的な例：アプリケーション設定管理

    public sealed class AppSettings
{
    private static readonly Lazy<AppSettings> _instance = 
        new Lazy<AppSettings>(() => new AppSettings());

    private readonly Dictionary<string, string> _settings;

    private AppSettings()
    {
        // 設定ファイルから読み込み（例）
        _settings = new Dictionary<string, string>
        {
            ["ApiUrl"] = "https://api.example.com",
            ["MaxRetries"] = "3",
            ["TimeoutSeconds"] = "30",
            ["LogLevel"] = "Info"
        };
        Console.WriteLine("アプリケーション設定を読み込みました");
    }

    public static AppSettings Instance => _instance.Value;

    public string GetSetting(string key)
    {
        return _settings.TryGetValue(key, out var value) ? value : string.Empty;
    }

    public T GetSetting<T>(string key) where T : IConvertible
    {
        var value = GetSetting(key);
        if (string.IsNullOrEmpty(value))
            return default(T);

        try
        {
            return (T)Convert.ChangeType(value, typeof(T));
        }
        catch
        {
            return default(T);
        }
    }
}

    使用例とテスト

    class SingletonExample
{
    static void Main()
    {
        Console.WriteLine("=== Singletonパターンのテスト ===");

        // 複数回取得しても同じインスタンス
        var db1 = DatabaseConnection.Instance;
        var db2 = DatabaseConnection.Instance;

        Console.WriteLine($"同じインスタンス: {ReferenceEquals(db1, db2)}");

        db1.ExecuteQuery("SELECT * FROM Users");
        db2.ExecuteQuery("SELECT * FROM Products");

        // 設定の取得テスト
        var settings = AppSettings.Instance;
        var apiUrl = settings.GetSetting("ApiUrl");
        var maxRetries = settings.GetSetting<int>("MaxRetries");

        Console.WriteLine($"API URL: {apiUrl}");
        Console.WriteLine($"最大リトライ回数: {maxRetries}");

        // 同じインスタンスかチェック
        var settings2 = AppSettings.Instance;
        Console.WriteLine($"設定も同じインスタンス: {ReferenceEquals(settings, settings2)}");
    }
}

    2. Factoryパターン - オブジェクト生成の複雑さを隠蔽

    パターンの目的

    「どのクラスのインスタンスを作るか」をクライアントから隠し、オブジェクト生成を一元管理します。

    Factory Methodパターンの実装

    // 製品の抽象クラス
public abstract class Logger
{
    public abstract void Log(string message);
    public abstract void LogError(string message);
}

// 具象製品：ファイルログ
public class FileLogger : Logger
{
    private readonly string _filePath;

    public FileLogger(string filePath)
    {
        _filePath = filePath;
        Console.WriteLine($"ファイルログ初期化: {_filePath}");
    }

    public override void Log(string message)
    {
        var logEntry = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] INFO: {message}";
        File.AppendAllText(_filePath, logEntry + Environment.NewLine);
        Console.WriteLine($"ファイルに記録: {message}");
    }

    public override void LogError(string message)
    {
        var logEntry = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] ERROR: {message}";
        File.AppendAllText(_filePath, logEntry + Environment.NewLine);
        Console.WriteLine($"エラーをファイルに記録: {message}");
    }
}

// 具象製品：コンソールログ
public class ConsoleLogger : Logger
{
    public ConsoleLogger()
    {
        Console.WriteLine("コンソールログ初期化");
    }

    public override void Log(string message)
    {
        Console.ForegroundColor = ConsoleColor.Green;
        Console.WriteLine($"[{DateTime.Now:HH:mm:ss}] INFO: {message}");
        Console.ResetColor();
    }

    public override void LogError(string message)
    {
        Console.ForegroundColor = ConsoleColor.Red;
        Console.WriteLine($"[{DateTime.Now:HH:mm:ss}] ERROR: {message}");
        Console.ResetColor();
    }
}

// 具象製品：データベースログ
public class DatabaseLogger : Logger
{
    private readonly string _connectionString;

    public DatabaseLogger(string connectionString)
    {
        _connectionString = connectionString;
        Console.WriteLine("データベースログ初期化");
    }

    public override void Log(string message)
    {
        // 実際のDBアクセスの代わりにシミュレート
        Console.WriteLine($"DB保存: {message}");
    }

    public override void LogError(string message)
    {
        Console.WriteLine($"DBにエラー保存: {message}");
    }
}

// Factory（生成の責任を持つ）
public static class LoggerFactory
{
    public static Logger CreateLogger(string type, string parameter = null)
    {
        return type.ToLower() switch
        {
            "file" => new FileLogger(parameter ?? "app.log"),
            "console" => new ConsoleLogger(),
            "database" => new DatabaseLogger(parameter ?? "DefaultConnection"),
            _ => throw new ArgumentException($"不明なログタイプ: {type}")
        };
    }

    // より高度なFactory：環境に基づく生成
    public static Logger CreateLoggerForEnvironment(string environment)
    {
        return environment.ToLower() switch
        {
            "development" => new ConsoleLogger(),
            "testing" => new FileLogger("test.log"),
            "production" => new DatabaseLogger("ProductionDB"),
            _ => new ConsoleLogger() // デフォルト
        };
    }
}

    3. Observerパターン - オブジェクト間の疎結合な通知

    パターンの目的

    「あるオブジェクトの状態変化を、複数の関連オブジェクトに自動的に通知する」仕組みを提供します。

    C#のイベント機能を使った実装

    // Subject（監視される側）
public class TemperatureSensor
{
    private double _temperature;

    // C#のイベント機能を使用
    public event Action<double> TemperatureChanged;
    public event Action<double> HighTemperatureAlert;
    public event Action<double> LowTemperatureAlert;

    public double Temperature
    {
        get => _temperature;
        set
        {
            if (Math.Abs(_temperature - value) > 0.1) // 0.1度以上の変化で通知
            {
                var oldTemperature = _temperature;
                _temperature = value;
                OnTemperatureChanged(value);
                CheckTemperatureAlerts(value);

                Console.WriteLine($"温度変化: {oldTemperature:F1}℃ → {value:F1}℃");
            }
        }
    }

    protected virtual void OnTemperatureChanged(double newTemperature)
    {
        TemperatureChanged?.Invoke(newTemperature);
    }

    private void CheckTemperatureAlerts(double temperature)
    {
        if (temperature > 30)
        {
            HighTemperatureAlert?.Invoke(temperature);
        }
        else if (temperature < 5)
        {
            LowTemperatureAlert?.Invoke(temperature);
        }
    }
}

// Observer1：温度表示
public class TemperatureDisplay
{
    private readonly string _name;

    public TemperatureDisplay(string name)
    {
        _name = name;
    }

    public void Subscribe(TemperatureSensor sensor)
    {
        sensor.TemperatureChanged += OnTemperatureChanged;
        sensor.HighTemperatureAlert += OnHighTemperature;
        sensor.LowTemperatureAlert += OnLowTemperature;
        Console.WriteLine($"{_name}が温度センサーを監視開始");
    }

    public void Unsubscribe(TemperatureSensor sensor)
    {
        sensor.TemperatureChanged -= OnTemperatureChanged;
        sensor.HighTemperatureAlert -= OnHighTemperature;
        sensor.LowTemperatureAlert -= OnLowTemperature;
        Console.WriteLine($"{_name}が温度センサーの監視を終了");
    }

    private void OnTemperatureChanged(double temperature)
    {
        Console.WriteLine($"  📺 {_name}: 現在温度 {temperature:F1}℃");
    }

    private void OnHighTemperature(double temperature)
    {
        Console.WriteLine($"  🔥 {_name}: 高温警告！ {temperature:F1}℃");
    }

    private void OnLowTemperature(double temperature)
    {
        Console.WriteLine($"  🧊 {_name}: 低温警告！ {temperature:F1}℃");
    }
}

    4. Strategyパターン - アルゴリズムの動的な切り替え

    パターンの目的

    「実行時にアルゴリズムを選択・変更できる」柔軟な設計を提供します。

    支払い処理の戦略パターン

    // 戦略インターフェース
public interface IPaymentStrategy
{
    PaymentResult ProcessPayment(decimal amount);
    string GetPaymentMethod();
    bool IsAvailable(decimal amount);
}

// 支払い結果
public class PaymentResult
{
    public bool IsSuccess { get; set; }
    public string Message { get; set; }
    public string TransactionId { get; set; }
    public DateTime ProcessedAt { get; set; }
}

// 具象戦略1：クレジットカード
public class CreditCardPayment : IPaymentStrategy
{
    private readonly string _cardNumber;
    private readonly string _cvv;
    private readonly decimal _limit;

    public CreditCardPayment(string cardNumber, string cvv, decimal limit = 100000)
    {
        _cardNumber = cardNumber;
        _cvv = cvv;
        _limit = limit;
    }

    public PaymentResult ProcessPayment(decimal amount)
    {
        Console.WriteLine($"クレジットカード決済処理中...");
        Thread.Sleep(1500); // 処理時間をシミュレート

        if (amount > _limit)
        {
            return new PaymentResult
            {
                IsSuccess = false,
                Message = $"利用限度額（{_limit:C}）を超過しています",
                ProcessedAt = DateTime.Now
            };
        }

        Console.WriteLine($"カード番号: {_cardNumber.Substring(0, 4)}****");
        return new PaymentResult
        {
            IsSuccess = true,
            Message = $"{amount:C}の決済が完了しました",
            TransactionId = $"CC{DateTime.Now:yyyyMMddHHmmss}",
            ProcessedAt = DateTime.Now
        };
    }

    public string GetPaymentMethod() => "クレジットカード";

    public bool IsAvailable(decimal amount) => amount <= _limit;
}

// コンテキスト：ショッピングカート
public class ShoppingCart
{
    private readonly List<(string Item, decimal Price)> _items = new();
    private IPaymentStrategy _paymentStrategy;

    public void AddItem(string item, decimal price)
    {
        _items.Add((item, price));
        Console.WriteLine($"商品追加: {item} ({price:C})");
    }

    public void SetPaymentStrategy(IPaymentStrategy strategy)
    {
        _paymentStrategy = strategy;
        Console.WriteLine($"支払い方法を設定: {strategy.GetPaymentMethod()}");
    }

    public decimal GetTotal()
    {
        return _items.Sum(item => item.Price);
    }

    public PaymentResult Checkout()
    {
        var total = GetTotal();

        if (_items.Count == 0)
        {
            Console.WriteLine("カートが空です");
            return new PaymentResult { IsSuccess = false, Message = "商品がありません" };
        }

        if (_paymentStrategy == null)
        {
            Console.WriteLine("支払い方法が設定されていません");
            return new PaymentResult { IsSuccess = false, Message = "支払い方法未設定" };
        }

        var result = _paymentStrategy.ProcessPayment(total);

        if (result.IsSuccess)
        {
            Console.WriteLine($"✅ {result.Message}");
            Console.WriteLine($"取引ID: {result.TransactionId}");
            _items.Clear(); // カートをクリア
        }
        else
        {
            Console.WriteLine($"❌ 決済失敗: {result.Message}");
        }

        return result;
    }
}

    5. Commandパターン - 操作のカプセル化

    パターンの目的

    「リクエストをオブジェクトとしてカプセル化」し、アンドゥ/リドゥ、キューイング、ログ記録を可能にします。

    テキストエディタの実装

    // Commandインターフェース
public interface ICommand
{
    void Execute();
    void Undo();
    string GetDescription();
}

// Receiver：実際の処理を行うクラス
public class TextEditor
{
    private StringBuilder _content = new StringBuilder();

    public string Content => _content.ToString();

    public void InsertText(int position, string text)
    {
        if (position < 0 || position > _content.Length)
            throw new ArgumentOutOfRangeException(nameof(position));

        _content.Insert(position, text);
    }

    public void DeleteText(int position, int length)
    {
        if (position < 0 || position + length > _content.Length)
            throw new ArgumentOutOfRangeException();

        _content.Remove(position, length);
    }

    public void ShowContent()
    {
        Console.WriteLine($"エディタ内容: \"{Content}\"");
    }
}

// 具象Command：テキスト挿入
public class InsertTextCommand : ICommand
{
    private readonly TextEditor _editor;
    private readonly int _position;
    private readonly string _text;

    public InsertTextCommand(TextEditor editor, int position, string text)
    {
        _editor = editor;
        _position = position;
        _text = text;
    }

    public void Execute()
    {
        _editor.InsertText(_position, _text);
    }

    public void Undo()
    {
        _editor.DeleteText(_position, _text.Length);
    }

    public string GetDescription() => $"挿入: \"{_text}\" at {_position}";
}

// Invoker：コマンドを管理・実行するクラス
public class EditorInvoker
{
    private readonly Stack<ICommand> _undoStack = new();
    private readonly Stack<ICommand> _redoStack = new();

    public void ExecuteCommand(ICommand command)
    {
        try
        {
            command.Execute();
            _undoStack.Push(command);
            _redoStack.Clear(); // 新しいコマンド実行時はRedoスタックをクリア

            Console.WriteLine($"実行: {command.GetDescription()}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"コマンド実行失敗: {ex.Message}");
        }
    }

    public void Undo()
    {
        if (_undoStack.Count > 0)
        {
            var command = _undoStack.Pop();
            try
            {
                command.Undo();
                _redoStack.Push(command);
                Console.WriteLine($"アンドゥ: {command.GetDescription()}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"アンドゥ失敗: {ex.Message}");
                _undoStack.Push(command);
            }
        }
        else
        {
            Console.WriteLine("アンドゥできる操作がありません");
        }
    }

    public void Redo()
    {
        if (_redoStack.Count > 0)
        {
            var command = _redoStack.Pop();
            try
            {
                command.Execute();
                _undoStack.Push(command);
                Console.WriteLine($"リドゥ: {command.GetDescription()}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"リドゥ失敗: {ex.Message}");
                _redoStack.Push(command);
            }
        }
        else
        {
            Console.WriteLine("リドゥできる操作がありません");
        }
    }
}

    Observer 状態変化の通知 イベント処理、MVC 疎結合、動的な関係 デバッグの困難さ、循環参照 ⭐⭐⭐
    Strategy アルゴリズムの切り替え 支払い処理、ソート 実行時切り替え、テスト容易 クラス数の増加 ⭐⭐

    使い分けの判断フローチャート

    問題の性質を分析
├─ システム全体で1つのインスタンスが必要
│  └─→ Singleton パターン
├─ オブジェクトの生成が複雑・条件分岐が多い
│  └─→ Factory パターン
├─ 状態変化を複数のオブジェクトに通知が必要
│  └─→ Observer パターン
├─ 実行時にアルゴリズムを切り替えたい
│  └─→ Strategy パターン
└─ 操作の記録・取り消し・再実行が必要
   └─→ Command パターン

    パフォーマンスへの影響と最適化

    メモリ使用量の考慮

    // Commandパターンでのメモリ最適化
public class CommandHistory
{
    private readonly Queue<ICommand> _commands = new();
    private const int MaxHistorySize = 100;

    public void AddCommand(ICommand command)
    {
        _commands.Enqueue(command);

        // 古いコマンドを削除してメモリを節約
        while (_commands.Count > MaxHistorySize)
        {
            _commands.Dequeue();
        }
    }
}

    Observerパターンでのメモリリーク防止

    // WeakReferenceを使用したObserver
public class WeakObserver<T>
{
    private readonly WeakReference<T> _target;
    private readonly Action<object> _callback;

    public WeakObserver(T target, Action<object> callback)
    {
        _target = new WeakReference<T>(target);
        _callback = callback;
    }

    public bool TryNotify(object data)
    {
        if (_target.TryGetTarget(out var target))
        {
            _callback(data);
            return true;
        }
        return false; // ターゲットがGCされた
    }
}

    よくある実装の間違いと対策

    1. Singletonの間違った実装

    // ❌ 悪い例：スレッドセーフでない
public class BadSingleton
{
    private static BadSingleton _instance;

    public static BadSingleton Instance
    {
        get
        {
            if (_instance == null) // 複数スレッドで問題
                _instance = new BadSingleton();
            return _instance;
        }
    }
}

// ✅ 良い例：Lazy<T>使用
public sealed class GoodSingleton
{
    private static readonly Lazy<GoodSingleton> _instance = 
        new Lazy<GoodSingleton>(() => new GoodSingleton());

    public static GoodSingleton Instance => _instance.Value;
}

    2. Observerパターンでの循環参照

    // ❌ 悪い例：循環参照が発生
public class Subject
{
    private readonly List<IObserver> _observers = new();

    public void AddObserver(IObserver observer)
    {
        _observers.Add(observer); // 強参照でメモリリーク
    }
}

// ✅ 良い例：適切な購読解除
public class Subject : IDisposable
{
    public event Action<Data> DataChanged;

    public void Dispose()
    {
        DataChanged = null; // イベントハンドラーをクリア
    }
}

    3. Strategyパターンでの戦略選択ロジック

    // ❌ 悪い例：if-else の連続
public IStrategy SelectStrategy(string type)
{
    if (type == "A") return new StrategyA();
    if (type == "B") return new StrategyB();
    if (type == "C") return new StrategyC();
    // 新しい戦略追加のたびに修正が必要
}

// ✅ 良い例：ファクトリーとの組み合わせ
public class StrategyFactory
{
    private readonly Dictionary<string, Func<IStrategy>> _strategies;

    public StrategyFactory()
    {
        _strategies = new Dictionary<string, Func<IStrategy>>
        {
            ["A"] = () => new StrategyA(),
            ["B"] = () => new StrategyB(),
            ["C"] = () => new StrategyC()
        };
    }

    public IStrategy CreateStrategy(string type)
    {
        return _strategies.TryGetValue(type, out var factory) 
            ? factory() 
            : throw new ArgumentException($"Unknown strategy: {type}");
    }
}

    実践での活用指針

    1. チーム開発での導入

    • 段階的導入: まず1つのパターンから始める
    • コードレビュー: パターンの適用が適切かチェック
    • ドキュメント化: パターンの使用理由を明記
    • 研修: チーム全体でパターンの理解を共有

    2. レガシーコードでの活用

    • リファクタリング: 段階的にパターンを適用
    • ラッパー: 既存コードをパターンでラップ
    • テスト追加: パターン適用前後でテストを追加

    3. パフォーマンス要件との両立

    • プロファイリング: パターン適用によるオーバーヘッドを測定
    • 最適化: 必要に応じてパターンを簡略化
    • トレードオフ: 保守性とパフォーマンスのバランス

    まとめ

    デザインパターンは強力なツールですが、適切な場面で適切に使用することが重要です。

    成功の鍵

    1. 問題の本質理解: パターンありきではなく、問題から始める
    2. 段階的適用: 小さく始めて徐々に拡張
    3. 継続的改善: 使用経験を積んでパターンを洗練
    4. チーム共有: パターンの知識をチーム全体で共有

    避けるべき罠

    • 過度な設計: 必要以上に複雑なパターンの適用
    • パターンの乱用: すべてをパターンで解決しようとする
    • 文脈無視: 問題の特性を考慮しないパターン選択

    適切に使用されたデザインパターンは、コードの品質向上、開発効率の改善、そしてチーム開発の円滑化に大きく貢献します。継続的な学習と実践を通じて、デザインパターンを効果的に活用できるエンジニアを目指しましょう。
